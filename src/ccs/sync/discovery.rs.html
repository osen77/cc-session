24/// Discover all conversation sessions in Claude Code history
25///
26/// When multiple files share the same session ID (e.g., main conversation and agent
27/// subprocess files), this function deduplicates by keeping the one with the most
28/// messages. This prevents agent files from overwriting main conversation files
29/// during merge operations.
30pub(crate) fn discover_sessions(
31    base_path: &Path,
32    filter: &FilterConfig,
33) -> Result<Vec<ConversationSession>> {
34    let candidates = collect_session_candidates(base_path, filter);
35
36    // Parsing dominates the cost and each file is independent; parse in parallel.
37    // par_iter + collect preserves candidate order, so dedup tie-breaking below
38    // behaves exactly as the previous serial loop.
39    use rayon::prelude::*;
40    let sessions: Vec<ConversationSession> = candidates
41        .par_iter()
42        .filter_map(|path| match ConversationSession::from_file(path) {
43            Ok(session) => Some(session),
44            Err(_) => {
45                log::warn!(target: crate::logger::SCAN_DIAGNOSTICS_TARGET, "legacy Claude session discovery skipped an unparseable file");
46                None
47            }
48        })
49        .collect();
50
51    // Prefer the canonical `<session_id>.jsonl` over KeepBoth/conflict copies.
52    // Within the same class, keep the entry with more messages so main sessions
53    // still beat smaller agent files.
54    fn is_canonical_session_file(session: &ConversationSession) -> bool {
55        Path::new(&session.file_path)
56            .file_stem()
57            .and_then(|stem| stem.to_str())
58            == Some(session.session_id.as_str())
59    }
60
61    let mut session_map: HashMap<String, ConversationSession> = HashMap::new();
62    for session in sessions {
63        session_map
64            .entry(session.session_id.clone())
65            .and_modify(|existing| {
66                let incoming_canonical = is_canonical_session_file(&session);
67                let existing_canonical = is_canonical_session_file(existing);
68                let replace = (incoming_canonical && !existing_canonical)
69                    || (incoming_canonical == existing_canonical
70                        && session.message_count() > existing.message_count());
71                if replace {
72                    *existing = session.clone();
73                }
74            })
75            .or_insert(session);
76    }
77
78    Ok(session_map.into_values().collect())
79}

81/// Walk `base_path` and return the validated `.jsonl` candidates, applying the
82/// same filter and path-security checks as `discover_sessions`.
83fn collect_session_candidates(base_path: &Path, filter: &FilterConfig) -> Vec<PathBuf> {
84    let mut candidates = Vec::new();
85    for entry in WalkDir::new(base_path).follow_links(false).into_iter() {
86        let Some(entry) = legacy_walk_entry(entry, "claude") else {
87            continue;
88        };
89        let path = entry.path();
90        if path.extension().and_then(|s| s.to_str()) != Some("jsonl") {
91            continue;
92        }
93        if !filter.should_include(path) {
94            continue;
95        }
96        if validate_regular_candidate(base_path, path).is_err() {
97            continue;
98        }
99        candidates.push(path.to_path_buf());
100    }
101    candidates
102}

104/// Count distinct sessions without fully parsing each file.
105///
106/// Mirrors `discover_sessions`' walk, filter and dedup semantics, but reads each
107/// file only until its session ID is found (usually the first line), falling back
108/// to the filename stem like the full parser does. Files that error before a
109/// session ID is found are skipped, matching discovery's skip of unparseable
110/// files. Intended for display counts (`ccs status`); sync flows that need the
111/// session content must keep using `discover_sessions`.
112pub(crate) fn count_unique_sessions(base_path: &Path, filter: &FilterConfig) -> Result<usize> {
113    let mut ids: std::collections::HashSet<String> = std::collections::HashSet::new();
114    for path in collect_session_candidates(base_path, filter) {
115        match crate::parser::first_entry_value(&path, |entry| entry.session_id.clone()) {
116            Ok(Some(id)) => {
117                ids.insert(id);
118            }
119            Ok(None) => {
120                if let Some(stem) = path.file_stem().and_then(|s| s.to_str()) {
121                    ids.insert(stem.to_string());
122                }
123            }
124            Err(_) => {
125                log::warn!(target: crate::logger::SCAN_DIAGNOSTICS_TARGET, "session count skipped an unreadable file");
126            }
127        }
128    }
129    Ok(ids.len())
130}

132/// Check for large conversation files and emit warnings
133///
134/// This helps users identify conversations that may be bloated with excessive
135/// file history, token usage, or other data. Large conversations can slow down
136/// sync operations and consume significant disk space.
137///
138/// # Arguments
139/// * `file_paths` - Iterator of file paths to check
140pub(crate) fn warn_large_files<P, I>(file_paths: I)
141where
142    P: AsRef<Path>,
143    I: IntoIterator<Item = P>,
144{
145    for path in file_paths {
146        let path = path.as_ref();
147
148        if let Ok(metadata) = fs::metadata(path) {
149            let size = metadata.len();
150
151            if size >= LARGE_FILE_WARNING_THRESHOLD {
152                let size_mb = size as f64 / (1024.0 * 1024.0);
153                println!(
154                    "  {} Large conversation file detected: {} ({:.1} MB)",
155                    "⚠️ ".yellow().bold(),
156                    path.file_name()
157                        .and_then(|n| n.to_str())
158                        .unwrap_or("unknown"),
159                    size_mb
160                );
161                println!(
162                    "     {}",
163                    "Consider archiving or cleaning up this conversation to improve sync performance"
164                        .dimmed()
165                );
166            }
167        }
168    }
169}

171/// Extract project name from Claude's encoded project directory name.
172///
173/// Claude encodes project paths by replacing '/' with '-', so a project at
174/// `/Users/abc/Documents/GitHub/myproject` becomes `-Users-abc-Documents-GitHub-myproject`.
175/// This function extracts the last segment (the actual project name).
176///
177/// # Examples
178/// - Input: `-Users-abc-Documents-GitHub-myproject` -> Output: `myproject`
179/// - Input: `myproject` -> Output: `myproject`
180/// - Input: `-root-projects-test` -> Output: `test`
181pub fn extract_project_name(encoded_path: &str) -> &str {
182    // The encoded path uses '-' as separator (from path encoding)
183    // Take the last non-empty segment
184    encoded_path
185        .rsplit('-')
186        .find(|s| !s.is_empty())
187        .unwrap_or(encoded_path)
188}

190/// Find a local Claude project directory that matches the given project name.
191///
192/// Scans `~/.claude/projects/` for directories that match the specified project name.
193/// Uses two strategies and merges results:
194/// 1. Fast: extract project name from encoded directory name (unreliable for hyphenated names)
195/// 2. Precise: read JSONL files and extract project name from `cwd` field
196///
197/// # Returns
198/// - `Some(PathBuf)` if exactly one matching project directory is found
199/// - `None` if no match found or multiple matches (ambiguous)
200pub fn find_local_project_by_name(
201    claude_projects_dir: &Path,
202    project_name: &str,
203) -> Option<PathBuf> {
204    validate_directory_root(claude_projects_dir).ok()?;
205    let entries: Vec<_> = std::fs::read_dir(claude_projects_dir)
206        .ok()?
207        .filter_map(|entry| entry.ok())
208        .filter(|entry| {
209            let path = entry.path();
210            matches!(
211                std::fs::symlink_metadata(&path),
212                Ok(metadata) if metadata.is_dir() && !metadata.file_type().is_symlink()
213            ) && validate_directory_candidate(claude_projects_dir, &path).is_ok()
214        })
215        .collect();
216
217    // Pass 1 (fast): match by encoded directory name.
218    // Skip directories whose name ends with '-' : Claude Code encodes each
219    // non-ASCII char (e.g. Chinese) as a single '-', so a dir like
220    // "-Users-mini-Documents-Projects-----" (cwd .../Projects/安装环境) ends with
221    // dashes and extract_project_name() would misread the PARENT segment
222    // ("Projects") as the project name, causing a false collision with a real
223    // sibling of that name. Such dirs are left to the precise Pass 2 (cwd).
224    let matches_from_dir: Vec<PathBuf> = entries
225        .iter()
226        .filter(|e| {
227            e.file_name().to_str().is_some_and(|name| {
228                !name.ends_with('-') && extract_project_name(name) == project_name
229            })
230        })
231        .map(|e| e.path())
232        .collect();
233
234    // Pass 2 (precise): read JSONL cwd to get real project name
235    // Always run - extract_project_name is unreliable for hyphenated names
236    let mut matches_from_cwd: Vec<PathBuf> = Vec::new();
237    for entry in &entries {
238        let dir_path = entry.path();
239        if let Some(real_name) = get_project_name_from_dir(&dir_path) {
240            if real_name == project_name {
241                matches_from_cwd.push(dir_path);
242            }
243        }
244    }
245
246    // Merge: cwd matches are authoritative, supplement with dir-name matches
247    let mut all_matches = matches_from_cwd;
248    for path in matches_from_dir {
249        if !all_matches.contains(&path) {
250            all_matches.push(path);
251        }
252    }
253
254    match all_matches.len() {
255        1 => Some(all_matches.into_iter().next().unwrap()),
256        n if n > 1 => {
257            log::warn!(
258                "Ambiguous match: {} local directories match project '{}': {:?}",
259                n,
260                project_name,
261                all_matches
262            );
263            None
264        }
265        _ => None,
266    }
267}

269/// Extract the real project name from a local project directory by reading its JSONL files.
270fn get_project_name_from_dir(dir_path: &Path) -> Option<String> {
271    validate_directory_root(dir_path).ok()?;
272    let files = std::fs::read_dir(dir_path).ok()?;
273    for file_entry in files.filter_map(|file| file.ok()) {
274        let file_path = file_entry.path();
275        if file_path.extension().and_then(|s| s.to_str()) != Some("jsonl") {
276            continue;
277        }
278        let Ok(metadata) = std::fs::symlink_metadata(&file_path) else {
279            continue;
280        };
281        if metadata.file_type().is_symlink() || !metadata.is_file() {
282            continue;
283        }
284        if validate_regular_candidate(dir_path, &file_path).is_err() {
285            continue;
286        }
287        // Stream only until the first cwd: these files can be hundreds of MB and
288        // a full parse here just to read one field dominated push startup.
289        if let Ok(Some(name)) = crate::parser::first_entry_value(&file_path, |entry| {
290            entry
291                .cwd
292                .as_deref()
293                .and_then(crate::parser::project_name_from_cwd)
294                .map(str::to_string)
295        }) {
296            return Some(name);
297        }
298    }
299    None
300}

302/// Get all project directories in Claude's projects folder that would map to the same project name.
303/// Used for collision detection when `use_project_name_only` is enabled.
304pub fn find_colliding_projects(
305    claude_projects_dir: &Path,
306) -> std::collections::HashMap<String, Vec<PathBuf>> {
307    use std::collections::HashMap;
308
309    let mut collisions: HashMap<String, Vec<PathBuf>> = HashMap::new();
310
311    if validate_directory_root(claude_projects_dir).is_err() {
312        return collisions;
313    }
314    if let Ok(entries) = std::fs::read_dir(claude_projects_dir) {
315        for entry in entries.filter_map(|entry| entry.ok()) {
316            let path = entry.path();
317            let Ok(metadata) = std::fs::symlink_metadata(&path) else {
318                continue;
319            };
320            if metadata.file_type().is_symlink()
321                || !metadata.is_dir()
322                || validate_directory_candidate(claude_projects_dir, &path).is_err()
323            {
324                continue;
325            }
326
327            // Prefer real project name from JSONL cwd, fall back to dir name extraction.
328            // Skip the fallback when the dir name ends with '-': a non-ASCII
329            // project name (e.g. Chinese) encodes to trailing dashes, and
330            // extract_project_name would misread the PARENT segment as the
331            // project name, producing a false collision with a real sibling.
332            // Without a reliable name such a dir cannot be grouped at all.
333            let project_name = get_project_name_from_dir(&path).unwrap_or_else(|| {
334                path.file_name()
335                    .and_then(|n| n.to_str())
336                    .filter(|n| !n.ends_with('-'))
337                    .map(|n| extract_project_name(n).to_string())
338                    .unwrap_or_default()
339            });
340            if !project_name.is_empty() {
341                collisions.entry(project_name).or_default().push(path);
342            }
343        }
344    }
345
346    // Only keep entries with more than one project (actual collisions)
347    collisions.retain(|_, paths| paths.len() > 1);
348    collisions
349}

351/// Result of checking sync repo directory structure consistency
352#[derive(Debug)]
353#[allow(dead_code)]
354pub struct DirectoryStructureCheck {
355    /// Directories using full path format (e.g., -Users-abc-project)
356    pub full_path_dirs: Vec<String>,
357    /// Directories using project name only format (e.g., project)
358    pub project_name_dirs: Vec<String>,
359    /// Whether the structure is consistent with the given config
360    pub is_consistent: bool,
361    /// Warning message if inconsistent
362    pub warning: Option<String>,
363}
364
365/// Check if the sync repo directory structure is consistent with the current config.
366///
367/// This helps detect when the user has switched modes and may have mixed directory formats.
368///
369/// # Arguments
370/// * `sync_repo_projects_dir` - Path to the projects directory in the sync repo
371/// * `use_project_name_only` - The current config setting
372///
373/// # Returns
374/// A `DirectoryStructureCheck` with details about the directory structure
375pub fn check_directory_structure_consistency(
376    sync_repo_projects_dir: &Path,
377    use_project_name_only: bool,
378) -> DirectoryStructureCheck {
379    let mut full_path_dirs = Vec::new();
380    let mut project_name_dirs = Vec::new();
381
382    if let Ok(entries) = std::fs::read_dir(sync_repo_projects_dir) {
383        for entry in entries.filter_map(|e| e.ok()) {
384            let path = entry.path();
385            if path.is_dir() {
386                if let Some(dir_name) = path.file_name().and_then(|n| n.to_str()) {
387                    // Skip hidden directories
388                    if dir_name.starts_with('.') {
389                        continue;
390                    }
391
392                    // Check if it looks like a full path (starts with - and contains multiple -)
393                    // e.g., -Users-abc-Documents-project
394                    let dash_count = dir_name.matches('-').count();
395                    if dir_name.starts_with('-') && dash_count >= 3 {
396                        full_path_dirs.push(dir_name.to_string());
397                    } else {
398                        project_name_dirs.push(dir_name.to_string());
399                    }
400                }
401            }
402        }
403    }
404
405    let has_full_path = !full_path_dirs.is_empty();
406    let has_project_name = !project_name_dirs.is_empty();
407
408    // Determine consistency
409    let (is_consistent, warning) = if has_full_path && has_project_name {
410        // Mixed mode - always inconsistent
411        (
412            false,
413            Some(format!(
414                "检测到混合目录格式：{} 个完整路径格式，{} 个项目名格式。\n\
415                 这可能导致数据重复。建议清理或统一目录格式。",
416                full_path_dirs.len(),
417                project_name_dirs.len()
418            )),
419        )
420    } else if use_project_name_only && has_full_path && !has_project_name {
421        // Config says project-name-only but repo has full paths
422        (
423            false,
424            Some(format!(
425                "配置为「多设备同步」模式，但同步仓库中存在 {} 个完整路径格式的目录。\n\
426                 建议清理这些目录或切换回「单设备备份」模式。",
427                full_path_dirs.len()
428            )),
429        )
430    } else if !use_project_name_only && has_project_name && !has_full_path {
431        // Config says full path but repo has project names only
432        (
433            false,
434            Some(format!(
435                "配置为「单设备备份」模式，但同步仓库中存在 {} 个项目名格式的目录。\n\
436                 建议切换到「多设备同步」模式以保持一致。",
437                project_name_dirs.len()
438            )),
439        )
440    } else {
441        (true, None)
442    };
443
444    DirectoryStructureCheck {
445        full_path_dirs,
446        project_name_dirs,
447        is_consistent,
448        warning,
449    }
450}

452/// Get list of memory files that exist in a directory
453#[allow(dead_code)]
454pub fn list_memory_files(memory_dir: &Path) -> Vec<PathBuf> {
455    let mut files = Vec::new();
456
457    if let Ok(entries) = fs::read_dir(memory_dir) {
458        for entry in entries.filter_map(|e| e.ok()) {
459            if entry.file_type().map(|t| t.is_file()).unwrap_or(false) {

760    fn test_find_local_project_non_ascii_sibling_no_false_ambiguity() {
761        // Regression: a project whose name contains non-ASCII chars (e.g. Chinese
762        // "安装环境") is encoded by Claude Code as trailing dashes in the directory
763        // name. extract_project_name() then misreads the PARENT segment as the
764        // project name, falsely colliding with a real sibling project of that name.
765        //
766        // Real layout that triggered this on the user's machine:
767        //   -Users-mini-Documents-Projects            cwd=/Users/mini/Documents/Projects
768        //   -Users-mini-Documents-Projects-----       cwd=/Users/mini/Documents/Projects/安装环境
769        // Looking up "Projects" must resolve to the first dir only, NOT None.
770        let temp_dir = tempdir().unwrap();
771        let projects_dir = temp_dir.path();
772
773        let real_projects = projects_dir.join("-Users-mini-Documents-Projects");
774        fs::create_dir(&real_projects).unwrap();
775        create_session_with_cwd(&real_projects, "sess-1", "/Users/mini/Documents/Projects");
776
777        // Non-ASCII project name "安装环境" (4 chars) -> 4 trailing dashes after the
778        // separator dash, so the encoded dir name ends with "Projects-----".
779        let non_ascii_sibling = projects_dir.join("-Users-mini-Documents-Projects-----");
780        fs::create_dir(&non_ascii_sibling).unwrap();
781        create_session_with_cwd(
782            &non_ascii_sibling,
783            "sess-2",
784            "/Users/mini/Documents/Projects/安装环境",
785        );
786
787        let result = find_local_project_by_name(projects_dir, "Projects");
788        assert!(
789            result.is_some(),
790            "Non-ASCII sibling must not cause false ambiguity with parent-named project"
791        );
792        assert!(result.unwrap().ends_with("-Users-mini-Documents-Projects"));
