113fn revalidate_suppression_for_remote(
114    store: &StateStore,
115    identity: &SessionIdentity,
116    remote_fingerprint: &str,
117) -> anyhow::Result<SuppressionRevalidation> {
118    store.transaction(|locked| {
119        let Some(entry) =
120            crate::session_maintenance::maintenance_state_for(&locked.state, identity)
121        else {
122            return Ok(SuppressionRevalidation::NotSuppressed);
123        };
124        match suppression_for_remote(&locked.state, identity, remote_fingerprint) {
125            SuppressionDecision::NotSuppressed => Ok(SuppressionRevalidation::NotSuppressed),
126            SuppressionDecision::SkipSameRevision => Ok(SuppressionRevalidation::SkipSameRevision),
127            SuppressionDecision::RestoreNewRevision => {
128                Ok(SuppressionRevalidation::Restore(PendingSuppressionClear {
129                    identity: identity.clone(),
130                    expected_fingerprint: entry.fingerprint.clone(),
131                    expected_lifecycle: entry.lifecycle,
132                }))
133            }
134        }
135    })
136}

138fn clear_pending_suppression_after_outcome(
139    store: Option<&StateStore>,
140    pending: &HashMap<SessionIdentity, PendingSuppressionClear>,
141    identity: &SessionIdentity,
142    outcome: SuppressionApplyOutcome,
143) {
144    if !should_clear_suppression(outcome) {
145        return;
146    }
147    let (Some(store), Some(token)) = (store, pending.get(identity)) else {
148        return;
149    };
150    match store.transaction(|locked| {
151        let cleared = locked.state.clear_suppression_if_matches(
152            &token.identity,
153            &token.expected_fingerprint,
154            token.expected_lifecycle,
155        );
156        if cleared {
157            locked.persist()?;
158        }
159        Ok(cleared)
160    }) {
161        Ok(true) => {}
162        Ok(false) => log::debug!(
163            "Suppression clear skipped because maintenance state changed for {}",
164            identity.session_id
165        ),
166        Err(error) => log::warn!(
167            "Failed to clear suppression after successful session restore safely: {}",
168            error
169        ),
170    }
171}

190fn revalidate_unchanged_session(
191    discovery_local: &ConversationSession,
192    remote: &ConversationSession,
193    local_root: &Path,
194    relative: &Path,
195) -> Result<bool> {
196    let destination = prepare_local_session_destination(local_root, relative)?;
197    let outcome = match ConversationSession::from_file_with_report(&destination) {
198        Ok(outcome) => outcome,
199        Err(error) => {
200            log::warn!(
201                "Unchanged session revalidation failed for {}: {}",
202                destination.display(),
203                error
204            );
205            return Ok(false);
206        }
207    };
208    if outcome.malformed_lines > 0
209        || outcome.value.session_id != remote.session_id
210        || discovery_local.session_id != remote.session_id
211    {
212        return Ok(false);
213    }
214    let current_hash = outcome.value.content_hash();
215    Ok(current_hash == discovery_local.content_hash() && current_hash == remote.content_hash())
216}

261fn propagate_tombstones(local_projects_root: &Path, registry: &TombstoneRegistry) -> Result<usize> {
262    validate_directory_root(local_projects_root)?;
263    let mut propagated = 0;
264    let entries = match fs::read_dir(local_projects_root) {
265        Ok(entries) => entries,
266        Err(error) if error.kind() == std::io::ErrorKind::NotFound => return Ok(0),
267        Err(error) => return Err(error.into()),
268    };
269
270    for entry in entries {
271        let entry = entry?;
272        let local_project_dir = entry.path();
273        let metadata = fs::symlink_metadata(&local_project_dir)?;
274        if metadata.file_type().is_symlink() || !metadata.is_dir() {
275            log::warn!(
276                "Skipping unsafe local project while applying tombstones: {}",
277                local_project_dir.display()
278            );
279            continue;
280        }
281        if let Err(error) = validate_directory_candidate(local_projects_root, &local_project_dir) {
282            log::warn!(
283                "Skipping unsafe local project while applying tombstones: {}",
284                error
285            );
286            continue;
287        }
288
289        for file in fs::read_dir(&local_project_dir)? {
290            let file = file?;
291            let file_name = file.file_name();
292            let Some(name) = file_name.to_str() else {
293                continue;
294            };
295            if !name.ends_with(".jsonl") {
296                continue;
297            }
298
299            let Some(session_id) =
300                crate::session_model::claude_session_id_from_path(Path::new(name))
301            else {
302                continue;
303            };
304            if !registry.contains(&session_id) {
305                continue;
306            }
307
308            let file_path = file.path();
309            let metadata = fs::symlink_metadata(&file_path)?;
310            if metadata.file_type().is_symlink() || !metadata.is_file() {
311                log::warn!(
312                    "Skipping unsafe tombstone deletion candidate: {}",
313                    file_path.display()
314                );
315                continue;
316            }
317            validate_regular_candidate(local_projects_root, &file_path)?;
318            let relative = file_path
319                .strip_prefix(local_projects_root)
320                .with_context(|| {
321                    format!(
322                        "local session path is outside projects root: {}",
323                        file_path.display()
324                    )
325                })?;
326            let candidate = safe_join_within_root(local_projects_root, relative)?;
327            validate_regular_candidate(local_projects_root, &candidate)?;
328            fs::remove_file(&candidate)
329                .with_context(|| format!("failed to propagate tombstone for {session_id}"))?;
330            propagated += 1;
331            log::debug!("Propagated remote deletion: {}", candidate.display());
332        }
333    }
334
335    Ok(propagated)
336}

338fn sync_auto_memory_from_remote(
339    sync_repo_path: &Path,
340    remote_projects_dir: &Path,
341    local_projects_root: &Path,
342    use_project_name_only: bool,
343) -> Result<Vec<String>> {
344    validate_sync_projects_root(sync_repo_path, remote_projects_dir)?;
345    validate_directory_root(local_projects_root)?;
346    let mut synced_projects = Vec::new();
347
348    for entry in fs::read_dir(remote_projects_dir)? {
349        let entry = entry?;
350        let project_name = match entry.file_name().to_str() {
351            Some(name) if !name.starts_with('.') && !name.is_empty() => name.to_string(),
352            _ => continue,
353        };
354        validate_project_component(&project_name)?;
355
356        let project_relative = PathBuf::from(&project_name);
357        let sync_project_dir = safe_join_within_sync_projects_root(
358            sync_repo_path,
359            remote_projects_dir,
360            &project_relative,
361        )?;
362        let project_metadata = fs::symlink_metadata(&sync_project_dir)?;
363        if project_metadata.file_type().is_symlink() {
364            anyhow::bail!("remote auto-memory project path must not be a symlink");
365        }
366        if !project_metadata.is_dir() {
367            continue;
368        }
369        validate_directory_candidate(remote_projects_dir, &sync_project_dir)?;
370
371        let memory_relative = project_relative.join("memory");
372        let remote_memory_path = safe_join_within_sync_projects_root(
373            sync_repo_path,
374            remote_projects_dir,
375            &memory_relative,
376        )?;
377        let remote_memory_metadata = match fs::symlink_metadata(&remote_memory_path) {
378            Ok(metadata) => metadata,
379            Err(error) if error.kind() == std::io::ErrorKind::NotFound => continue,
380            Err(error) => return Err(error.into()),
381        };
382        if remote_memory_metadata.file_type().is_symlink() || !remote_memory_metadata.is_dir() {
383            anyhow::bail!("remote auto-memory path must be a non-symlink directory");
384        }
385        validate_directory_candidate(remote_projects_dir, &remote_memory_path)?;
386
387        let local_project_dir = if use_project_name_only {
388            find_local_project_by_name(local_projects_root, &project_name)
389        } else {
390            let local_path = safe_join_within_root(local_projects_root, &project_relative)?;
391            match fs::symlink_metadata(&local_path) {
392                Ok(metadata) if metadata.is_dir() && !metadata.file_type().is_symlink() => {
393                    Some(local_path)
394                }
395                Ok(_) => None,
396                Err(error) if error.kind() == std::io::ErrorKind::NotFound => None,
397                Err(error) => return Err(error.into()),
398            }
399        };
400
401        let Some(local_project_dir) = local_project_dir else {
402            log::debug!(
403                "No local project found for '{}', skipping memory sync",
404                project_name
405            );
406            continue;
407        };
408        validate_directory_candidate(local_projects_root, &local_project_dir)?;
409
410        let local_memory_path = safe_join_within_root(&local_project_dir, Path::new("memory"))?;
411        match fs::symlink_metadata(&local_memory_path) {
412            Ok(metadata) if metadata.file_type().is_symlink() || !metadata.is_dir() => {
413                anyhow::bail!("local auto-memory path must be a non-symlink directory");
414            }
415            Ok(_) => {}
416            Err(error) if error.kind() == std::io::ErrorKind::NotFound => {
417                fs::create_dir_all(&local_memory_path)?;
418            }
419            Err(error) => return Err(error.into()),
420        }
421        validate_directory_candidate(local_projects_root, &local_memory_path)?;
422
423        for memory_entry in fs::read_dir(&remote_memory_path)? {
424            let memory_entry = memory_entry?;
425            let source = memory_entry.path();
426            let source_metadata = fs::symlink_metadata(&source)?;
427            if source_metadata.file_type().is_symlink() {
428                anyhow::bail!("remote auto-memory contains a symlink file");
429            }
430            if !source_metadata.is_file() {
431                continue;
432            }
433            validate_regular_candidate(remote_projects_dir, &source)?;
434
435            let destination =
436                safe_join_within_root(&local_memory_path, Path::new(&memory_entry.file_name()))?;
437            if let Ok(metadata) = fs::symlink_metadata(&destination) {
438                if metadata.file_type().is_symlink() || !metadata.is_file() {
439                    anyhow::bail!("local auto-memory destination is not a regular file");
440                }
441            }
442
443            validate_regular_candidate(remote_projects_dir, &source)?;
444            let destination =
445                safe_join_within_root(&local_memory_path, Path::new(&memory_entry.file_name()))?;
446            fs::copy(&source, &destination)?;
447        }
448        synced_projects.push(project_name);
449    }
450
451    Ok(synced_projects)
452}

454/// Pull and merge history from sync repository
455pub fn pull_history(
456    fetch_remote: bool,
457    branch: Option<&str>,
458    interactive: bool,
459    verbosity: crate::VerbosityLevel,
460) -> Result<()> {
461    use crate::VerbosityLevel;
462
463    if verbosity != VerbosityLevel::Quiet {
464        println!("{}", "Pulling Claude Code history...".cyan().bold());
465    }
466
467    let state = SyncState::load()?;
468
469    // Serialize against any other process writing this repository — a pull
470    // rebases and rewrites the working tree, so it cannot overlap a push.
471    let Some(_repo_lock) = crate::sync::repo_lock::RepoLock::acquire_or_report(
472        &state.sync_repo_path,
473        "拉取",
474        verbosity,
475    )?
476    else {
477        return Ok(());
478    };
479
480    let repo = scm::open(&state.sync_repo_path)?;
481    let filter = FilterConfig::load()?;
482    let claude_dir = claude_projects_dir()?;
483    validate_directory_root(&claude_dir)?;
484
485    // Get the current branch name for operation record
486    let branch_name = branch
487        .map(|s| s.to_string())
488        .or_else(|| repo.current_branch().ok())
489        .unwrap_or_else(|| "main".to_string());
490
491    // Fetch from remote if configured
492    if fetch_remote && state.has_remote {
493        println!("  {} from remote...", "Fetching".cyan());
494
495        match repo.pull("origin", &branch_name) {
496            Ok(_) => println!("  {} Pulled from origin/{}", "✓".green(), branch_name),
497            Err(e) => {
498                log::warn!("Failed to pull: {}", e);
499                log::info!("Continuing with local sync repository state...");
500            }
501        }
502    }
503
504    // ============================================================================
505    // PROPAGATE INTENTIONAL DELETIONS
506    // ============================================================================
507    // Before discovering local sessions, check if the sync repo has any
508    // registered tombstones that we haven't applied locally yet.
509    match TombstoneRegistry::load(&state.sync_repo_path) {
510        Ok(registry) if !registry.is_empty() => {
511            if verbosity != VerbosityLevel::Quiet {
512                println!("  {} tombstones...", "Checking".cyan());
513            }
514            let propagated_deletes = propagate_tombstones(&claude_dir, &registry)?;
515            if propagated_deletes > 0 && verbosity != VerbosityLevel::Quiet {
516                println!(
517                    "  {} Propagated {} intentional deletion(s) from other devices",
518                    "✓".green(),
519                    propagated_deletes
520                );
521            }
522        }
523        Ok(_) => {}
524        Err(error) => {
525            log::warn!("Failed to load tombstone registry safely: {}", error);
526        }
527    }
528
529    // Discover local sessions.
530    println!("  {} local sessions...", "Discovering".cyan());
531    let local_sessions = discover_sessions(&claude_dir, &filter)?;
532    println!(
533        "  {} {} local sessions",
534        "Found".green(),
535        local_sessions.len()
536    );
537
538    // Discover remote sessions
539    let remote_projects_dir = state.sync_repo_path.join(&filter.sync_subdirectory);
540    validate_sync_projects_root(&state.sync_repo_path, &remote_projects_dir)?;
541    println!("  {} remote sessions...", "Discovering".cyan());
542    let discovered_remote_sessions = discover_sessions(&remote_projects_dir, &filter)?;
543    let maintenance_store = ConfigManager::config_dir().ok().map(|config_dir| {
544        crate::session_maintenance::state::StateStore::from_config_dir(&config_dir)
545    });
546    let maintenance_state = match maintenance_store.as_ref().map(|store| store.load()) {
547        Some(Ok(state)) => Some(state),
548        Some(Err(error)) => {
549            log::warn!(
550                "Failed to load session maintenance state; restoring remote sessions safely: {}",
551                error
552            );
553            None
554        }
555        None => {
556            log::warn!(
557                "Failed to locate session maintenance state; restoring remote sessions safely"
558            );
559            None
560        }
561    };
562    let mut suppressed_remote_count = 0usize;
563    let mut pending_suppression_clears = HashMap::new();
564    let mut remote_sessions = Vec::with_capacity(discovered_remote_sessions.len());
565    for remote_session in discovered_remote_sessions {
566        let identity = SessionIdentity {
567            source: SessionSource::Claude,
568            session_id: remote_session.session_id.clone(),
569        };
570        let Some(state) = maintenance_state.as_ref() else {
571            remote_sessions.push(remote_session);
572            continue;
573        };
574        if crate::session_maintenance::maintenance_state_for(state, &identity).is_none() {
575            remote_sessions.push(remote_session);
576            continue;
577        }
578        let fingerprint = match fingerprint_file(Path::new(&remote_session.file_path)) {
579            Ok(fingerprint) => fingerprint.digest,
580            Err(error) => {
581                log::warn!(
582                    "Failed to fingerprint remote session safely; restoring it: {}",
583                    error
584                );
585                remote_sessions.push(remote_session);
586                continue;
587            }
588        };
589        match suppression_for_remote(state, &identity, &fingerprint) {
590            SuppressionDecision::SkipSameRevision => {
591                suppressed_remote_count += 1;
592            }
593            SuppressionDecision::RestoreNewRevision => {
594                let revalidated = maintenance_store
595                    .as_ref()
596                    .map(|store| revalidate_suppression_for_remote(store, &identity, &fingerprint));
597                match revalidated {
598                    Some(Ok(SuppressionRevalidation::SkipSameRevision)) => {
599                        suppressed_remote_count += 1;
600                    }
601                    Some(Ok(SuppressionRevalidation::Restore(token))) => {
602                        pending_suppression_clears.insert(identity, token);
603                        remote_sessions.push(remote_session);
604                    }
605                    Some(Ok(SuppressionRevalidation::NotSuppressed)) | None => {
606                        remote_sessions.push(remote_session);
607                    }
608                    Some(Err(error)) => {
609                        log::warn!(
610                            "Failed to revalidate remote suppression safely; restoring it without clearing state: {}",
611                            error
612                        );
613                        remote_sessions.push(remote_session);
614                    }
615                }
616            }
617            SuppressionDecision::NotSuppressed => remote_sessions.push(remote_session),
618        }
619    }
620    println!(
621        "  {} {} remote sessions",
622        "Found".green(),
623        remote_sessions.len()
624    );
625    let remote_revisions: HashMap<String, (PathBuf, String)> = remote_sessions
626        .iter()
627        .map(|session| {
628            let path = Path::new(&session.file_path);
629            let relative = path
630                .strip_prefix(&remote_projects_dir)
631                .context("remote session is outside sync projects root")?
632                .to_path_buf();
633            let fingerprint = fingerprint_file(path)?.digest;
634            Ok((session.session_id.clone(), (relative, fingerprint)))
635        })
636        .collect::<Result<_>>()?;
637    if suppressed_remote_count > 0 && verbosity != VerbosityLevel::Quiet {
638        println!(
639            "  {} Suppressed {} unchanged locally recycled session(s)",
640            "✓".green(),
641            suppressed_remote_count
642        );
643    }
644
645    // ============================================================================
646    // CONFLICT DETECTION (moved before snapshot for efficiency)
647    // ============================================================================
648    // Detect conflicts FIRST so we only backup files that will be modified
649    if verbosity != VerbosityLevel::Quiet {
650        println!("  {} conflicts...", "Detecting".cyan());
651    }
652    let mut detector = ConflictDetector::new();
653    detector.detect(&local_sessions, &remote_sessions);
654
655    // ============================================================================
656    // SHOW SUMMARY AND INTERACTIVE CONFIRMATION
657    // ============================================================================
658    if verbosity != VerbosityLevel::Quiet {
659        println!();
660        println!("{}", "Pull Summary:".bold().cyan());
661        println!("  {} Local sessions: {}", "•".cyan(), local_sessions.len());
662        println!(
663            "  {} Remote sessions: {}",
664            "•".cyan(),
665            remote_sessions.len()
666        );
667        println!();
668    }
669
670    if verbosity == VerbosityLevel::Verbose {
671        println!("{}", "Remote sessions to be pulled:".bold());

838                        "malformed or mismatched snapshot baseline".to_string(),
839                    );

852                        &pending_suppression_clears,

862            };

881                        .context("conflict destination is outside Claude projects root")?;

916                                SyncOperation::Conflict,
917                            ) {
918                                Ok(summary) => affected_conversations.push(summary),
919                                Err(error) => log::warn!(
920                                    "Failed to record append-only SmartMerge {}: {}",
921                                    conflict.session_id,
922                                    error
923                                ),
924                            }
925                        }
926                        Ok(AppendSessionWriteOutcome::Noop) => {
927                            smart_merge_success_count += 1;
928                            completed_sessions.insert(conflict.session_id.clone());
929                            incomplete_reasons.remove(&conflict.session_id);
930                            println!(
931                                "  {} Smart merge for {} required no local write ({}/{} UUIDs)",

934                                stats.emitted_uuid_count,
935                                stats.expected_uuid_count
936                            );

938                                maintenance_store.as_ref(),
939                                &pending_suppression_clears,
940                                &identity,
941                                SuppressionApplyOutcome::Unchanged,
942                            );
943                            match ConversationSummary::new(
944                                conflict.session_id.clone(),
945                                local_relative.to_string_lossy().to_string(),
946                                merged_session.latest_timestamp(),
947                                merged_session.message_count(),
948                                SyncOperation::Conflict,
949                            ) {
950                                Ok(summary) => affected_conversations.push(summary),
951                                Err(error) => log::warn!(
952                                    "Failed to record no-op SmartMerge {}: {}",
953                                    conflict.session_id,
954                                    error
955                                ),
956                            }
957                        }
958                        Ok(AppendSessionWriteOutcome::SkippedChanged) => {
959                            conflict.resolution = crate::conflict::ConflictResolution::Pending;
960                            log::warn!(
961                                "Append-only SmartMerge preconditions changed for {}; falling back to KeepBoth",
962                                conflict.session_id
963                            );
964                            clear_pending_suppression_after_outcome(
965                                maintenance_store.as_ref(),
966                                &pending_suppression_clears,
967                                &identity,
968                                SuppressionApplyOutcome::SkippedChanged,
969                            );
970                            smart_merge_failed_conflicts.push(conflict.clone());
971                        }
972                        Err(error) => {
973                            conflict.resolution = crate::conflict::ConflictResolution::Pending;
974                            log::warn!(
975                                "SmartMerge for {} requires rewriting existing entries or append failed; falling back to KeepBoth: {}",
976                                conflict.session_id,
977                                error
978                            );
979                            clear_pending_suppression_after_outcome(
980                                maintenance_store.as_ref(),
981                                &pending_suppression_clears,
982                                &identity,
983                                SuppressionApplyOutcome::WriteFailed,
984                            );
985                            smart_merge_failed_conflicts.push(conflict.clone());
986                        }
987                    }
988                }
989                Err(error) => {
990                    log::warn!("Smart merge failed for {}: {}", conflict.session_id, error);
991                    log::info!("Falling back to manual resolution...");
992                    clear_pending_suppression_after_outcome(
993                        maintenance_store.as_ref(),
994                        &pending_suppression_clears,
995                        &identity,
996                        SuppressionApplyOutcome::WriteFailed,
997                    );
998                    smart_merge_failed_conflicts.push(conflict.clone());
999                }
1000            }
1001        }

1003        println!(
1004            "  {} Successfully smart merged {}/{} conflicts",
1005            "✓".green(),
1006            smart_merge_success_count,
1007            detector.conflict_count()
1008        );
1009
1010        // If some smart merges failed, handle them with interactive/keep-both resolution
1011        let renames = if !smart_merge_failed_conflicts.is_empty() {
1012            println!(
1013                "  {} {} conflicts require manual resolution",
1014                "!".yellow(),
1015                smart_merge_failed_conflicts.len()
1016            );
1017
1018            let use_interactive = crate::interactive_conflict::is_interactive();
1019            let resolution_result = if use_interactive {
1020                println!(
1021                    "\n{} Running in interactive mode for remaining conflicts",
1022                    "→".cyan()
1023                );
1024                let baseline_map: HashMap<_, _> = baseline_sessions
1025                    .iter()
1026                    .map(|(session_id, session)| (session_id.clone(), session))
1027                    .collect();
1028                crate::interactive_conflict::resolve_conflicts_interactive_with_sessions(
1029                    &mut smart_merge_failed_conflicts,
1030                    Some(&baseline_map),
1031                    Some(&remote_map),
1032                )?
1033            } else {
1034                println!(
1035                    "\n{} Using automatic conflict resolution (keep both versions)",
1036                    "→".cyan()
1037                );
1038                let mut result = crate::interactive_conflict::ResolutionResult::new();
1039                result.keep_both = smart_merge_failed_conflicts.clone();
1040                result
1041            };
1042
1043            let applied = match crate::interactive_conflict::apply_resolutions_guarded(
1044                &resolution_result,
1045                &remote_sessions,
1046                &claude_dir,
1047                &remote_projects_dir,
1048                &conflict_baselines,
1049            ) {
1050                Ok(applied) => applied,
1051                Err(error) => {
1052                    log::warn!(
1053                        "Conflict fallback could not write a local remote copy; remote remains protected in the sync repository: {}",
1054                        error
1055                    );
1056                    for conflict in &smart_merge_failed_conflicts {
1057                        incomplete_reasons.insert(
1058                            conflict.session_id.clone(),
1059                            format!("conflict fallback write failed: {error}"),
1060                        );
1061                    }
1062                    crate::interactive_conflict::GuardedApplyResult::default()
1063                }
1064            };
1065            for (session_id, outcome) in &applied.outcomes {
1066                let identity = SessionIdentity {
1067                    source: SessionSource::Claude,
1068                    session_id: session_id.clone(),
1069                };
1070                let suppression_outcome = match outcome {
1071                    crate::interactive_conflict::GuardedApplyOutcome::Written { .. } => {
1072                        completed_sessions.insert(session_id.clone());
1073                        incomplete_reasons.remove(session_id);
1074                        SuppressionApplyOutcome::Written
1075                    }
1076                    crate::interactive_conflict::GuardedApplyOutcome::CreatedConflictCopy {
1077                        ..
1078                    } => {
1079                        incomplete_reasons.insert(
1080                            session_id.clone(),
1081                            "remote preserved as KeepBoth after incomplete merge".to_string(),
1082                        );
1083                        SuppressionApplyOutcome::WriteFailed
1084                    }
1085                    crate::interactive_conflict::GuardedApplyOutcome::SkippedChanged => {
1086                        skipped_changed_count += 1;
1087                        incomplete_reasons.insert(
1088                            session_id.clone(),
1089                            "local session changed during conflict resolution".to_string(),
1090                        );
1091                        println!(
1092                            "  {} Local session {} changed during conflict resolution; preserved local file",
1093                            "!".yellow(),
1094                            session_id
1095                        );
1096                        SuppressionApplyOutcome::SkippedChanged
1097                    }
1098                    crate::interactive_conflict::GuardedApplyOutcome::KeptLocal => {
1099                        completed_sessions.insert(session_id.clone());
1100                        incomplete_reasons.remove(session_id);
1101                        SuppressionApplyOutcome::Cancelled
1102                    }
1103                };
1104                clear_pending_suppression_after_outcome(
1105                    maintenance_store.as_ref(),
1106                    &pending_suppression_clears,
1107                    &identity,
1108                    suppression_outcome,
1109                );
1110            }
1111
1112            crate::interactive_conflict::update_reported_resolutions(
1113                detector.conflicts_mut(),
1114                &resolution_result,
1115                &applied,
1116            );
1117            let report = ConflictReport::from_conflicts(detector.conflicts());
1118            save_conflict_report(&report)?;
1119            applied.renames
1120        } else {
1121            // All conflicts resolved via smart merge
1122            Vec::new()
1123        };
1124
1125        // Track all conflicts in affected conversations
1126        for (_original_path, renamed_path) in &renames {
1127            let relative_path = renamed_path
1128                .strip_prefix(&claude_dir)
1129                .unwrap_or(renamed_path)
1130                .to_string_lossy()
1131                .to_string();
1132
1133            // Find the session ID from the renamed path
1134            if let Some(session) = remote_sessions.iter().find(|s| {
1135                let session_file = Path::new(&s.file_path).file_name();
1136                let renamed_file = renamed_path.file_name();
1137                // Try to match based on session ID in filename
1138                session_file
1139                    .and_then(|f| f.to_str())
1140                    .and_then(|name| name.split('-').next())
1141                    == renamed_file
1142                        .and_then(|f| f.to_str())
1143                        .and_then(|name| name.split('-').next())
1144            }) {
1145                match ConversationSummary::new(
1146                    session.session_id.clone(),
1147                    relative_path.clone(),
1148                    session.latest_timestamp(),
1149                    session.message_count(),
1150                    SyncOperation::Conflict,
1151                ) {
1152                    Ok(summary) => affected_conversations.push(summary),
1153                    Err(e) => log::warn!(
1154                        "Failed to create summary for conflict {}: {}",
1155                        relative_path,
1156                        e
1157                    ),
1158                }
1159            }
1160        }
1161
1162        println!(
1163            "\n{} View details with: {} report",
1164            "Hint:".cyan(),
1165            BINARY_NAME
1166        );
1167    } else {
1168        println!("  {} No conflicts detected", "✓".green());
1169    }

1172    // MERGE NON-CONFLICTING SESSIONS
1173    // ============================================================================
1174    println!("  {} non-conflicting sessions...", "Merging".cyan());
1175    let local_map: HashMap<_, _> = local_sessions
1176        .iter()
1177        .map(|s| (s.session_id.clone(), s))
1178        .collect();
1179
1180    let mut merged_count = 0;
1181    let mut added_count = 0;
1182    let mut unchanged_count = 0;
1183    let mut skipped_no_local_match = 0;
1184    let mut skipped_existing_target = 0;
1185
1186    for remote_session in &remote_sessions {
1187        // Skip if conflicts were detected
1188        if detector
1189            .conflicts()
1190            .iter()
1191            .any(|c| c.session_id == remote_session.session_id)
1192        {
1193            continue;
1194        }
1195
1196        let relative_path_for_tracking = if filter.use_project_name_only {
1197            // Extract project name and session filename from remote path.
1198            let remote_relative = Path::new(&remote_session.file_path)
1199                .strip_prefix(&remote_projects_dir)
1200                .ok()
1201                .unwrap_or_else(|| Path::new(&remote_session.file_path));
1202            let project_name = remote_relative
1203                .components()
1204                .next()
1205                .and_then(|component| component.as_os_str().to_str())
1206                .unwrap_or("unknown");
1207            validate_project_component(project_name)?;
1208
1209            let Some(local_project_dir) = find_local_project_by_name(&claude_dir, project_name)
1210            else {
1211                log::debug!(
1212                    "No matching local project found for '{}', skipping",
1213                    project_name
1214                );
1215                skipped_no_local_match += 1;
1216                incomplete_reasons.insert(
1217                    remote_session.session_id.clone(),
1218                    "no matching local project".to_string(),
1219                );
1220                let identity = SessionIdentity {
1221                    source: SessionSource::Claude,
1222                    session_id: remote_session.session_id.clone(),
1223                };
1224                clear_pending_suppression_after_outcome(
1225                    maintenance_store.as_ref(),
1226                    &pending_suppression_clears,
1227                    &identity,
1228                    SuppressionApplyOutcome::SkippedNoLocalProject,
1229                );
1230                continue;
1231            };
1232            validate_directory_candidate(&claude_dir, &local_project_dir)?;
1233
1234            let Some(filename) = remote_relative.file_name() else {
1235                log::warn!(
1236                    "Could not extract filename from remote path: {:?}",
1237                    remote_relative
1238                );
1239                skipped_no_local_match += 1;
1240                incomplete_reasons.insert(
1241                    remote_session.session_id.clone(),
1242                    "no matching local project".to_string(),
1243                );
1244                let identity = SessionIdentity {
1245                    source: SessionSource::Claude,
1246                    session_id: remote_session.session_id.clone(),
1247                };
1248                clear_pending_suppression_after_outcome(
1249                    maintenance_store.as_ref(),
1250                    &pending_suppression_clears,
1251                    &identity,
1252                    SuppressionApplyOutcome::SkippedNoLocalProject,
1253                );
1254                continue;
1255            };
1256            let local_project_relative = local_project_dir
1257                .strip_prefix(&claude_dir)
1258                .context("matched local project is outside Claude projects root")?;
1259            local_project_relative.join(filename)
1260        } else {
1261            Path::new(&remote_session.file_path)
1262                .strip_prefix(&remote_projects_dir)
1263                .context("remote session is outside the sync projects root")?
1264                .to_path_buf()
1265        };
1266
1267        if let Err(error) =
1268            prepare_local_session_destination(&claude_dir, &relative_path_for_tracking)
1269        {
1270            incomplete_reasons.insert(
1271                remote_session.session_id.clone(),
1272                format!("unsafe local destination: {error}"),
1273            );
1274            let identity = SessionIdentity {
1275                source: SessionSource::Claude,
1276                session_id: remote_session.session_id.clone(),
1277            };
1278            clear_pending_suppression_after_outcome(
1279                maintenance_store.as_ref(),
1280                &pending_suppression_clears,
1281                &identity,
1282                SuppressionApplyOutcome::WriteFailed,
1283            );
1284            continue;
1285        }
1286
1287        // Determine operation type based on local state
1288        let operation = if let Some(local) = local_map.get(&remote_session.session_id) {
1289            if local.content_hash() == remote_session.content_hash() {
1290                SyncOperation::Unchanged
1291            } else {
1292                SyncOperation::Modified
1293            }
1294        } else {
1295            SyncOperation::Added
1296        };
1297
1298        // A remote-only discovery result does not prove the destination is absent:
1299        // local filtering or parse errors may have hidden a physical file. Never
1300        // replace such a file through the Added path.
1301        if operation == SyncOperation::Added {
1302            match write_added_session_without_overwrite(
1303                remote_session,
1304                &claude_dir,
1305                &relative_path_for_tracking,
1306            ) {
1307                Ok(AddedSessionWriteOutcome::Written(_)) => {
1308                    added_count += 1;
1309                    merged_count += 1;
1310                    completed_sessions.insert(remote_session.session_id.clone());
1311                    incomplete_reasons.remove(&remote_session.session_id);
1312                }
1313                Ok(AddedSessionWriteOutcome::SkippedExisting(destination)) => {
1314                    log::warn!(
1315                        "Skipping remote session {} because its local destination already exists but was not discovered; preserving local file: {}",
1316                        remote_session.session_id,
1317                        destination.display()
1318                    );
1319                    skipped_existing_target += 1;
1320                    incomplete_reasons.insert(
1321                        remote_session.session_id.clone(),
1322                        "existing local target was not safely discoverable".to_string(),
1323                    );
1324                    let identity = SessionIdentity {
1325                        source: SessionSource::Claude,
1326                        session_id: remote_session.session_id.clone(),
1327                    };
1328                    clear_pending_suppression_after_outcome(
1329                        maintenance_store.as_ref(),
1330                        &pending_suppression_clears,
1331                        &identity,
1332                        SuppressionApplyOutcome::SkippedExistingTarget,
1333                    );
1334                    continue;
1335                }
1336                Err(error) => {
1337                    incomplete_reasons.insert(
1338                        remote_session.session_id.clone(),
1339                        format!("Added write failed: {error}"),
1340                    );
1341                    let identity = SessionIdentity {
1342                        source: SessionSource::Claude,
1343                        session_id: remote_session.session_id.clone(),
1344                    };
1345                    clear_pending_suppression_after_outcome(
1346                        maintenance_store.as_ref(),
1347                        &pending_suppression_clears,
1348                        &identity,
1349                        SuppressionApplyOutcome::WriteFailed,
1350                    );
1351                    continue;
1352                }
1353            }
1354        } else if operation == SyncOperation::Modified {
1355            // Any differing session present on both sides should have been a
1356            // conflict. Treat reaching this branch as an invariant failure and
1357            // preserve the local file rather than overwriting it.
1358            log::warn!(
1359                "Skipping non-conflict Modified session {} to preserve the local file",
1360                remote_session.session_id
1361            );
1362            skipped_changed_count += 1;
1363            incomplete_reasons.insert(
1364                remote_session.session_id.clone(),
1365                "unexpected non-conflict Modified classification".to_string(),
1366            );
1367            let identity = SessionIdentity {
1368                source: SessionSource::Claude,
1369                session_id: remote_session.session_id.clone(),
1370            };
1371            clear_pending_suppression_after_outcome(
1372                maintenance_store.as_ref(),
1373                &pending_suppression_clears,
1374                &identity,
1375                SuppressionApplyOutcome::SkippedChanged,
1376            );
1377            continue;
1378        }
1379
1380        if operation == SyncOperation::Unchanged {
1381            let Some(discovery_local) = local_map.get(&remote_session.session_id) else {
1382                unreachable!("Unchanged session must have a discovery baseline")
1383            };
1384            if !revalidate_unchanged_session(
1385                discovery_local,
1386                remote_session,
1387                &claude_dir,
1388                &relative_path_for_tracking,
1389            )? {
1390                skipped_changed_count += 1;
1391                incomplete_reasons.insert(
1392                    remote_session.session_id.clone(),
1393                    "local session changed before Unchanged revalidation".to_string(),
1394                );
1395                let identity = SessionIdentity {
1396                    source: SessionSource::Claude,
1397                    session_id: remote_session.session_id.clone(),
1398                };
1399                clear_pending_suppression_after_outcome(
1400                    maintenance_store.as_ref(),
1401                    &pending_suppression_clears,
1402                    &identity,
1403                    SuppressionApplyOutcome::SkippedChanged,
1404                );
1405                continue;
1406            }
1407            unchanged_count += 1;
1408            completed_sessions.insert(remote_session.session_id.clone());
1409            incomplete_reasons.remove(&remote_session.session_id);
1410        }
1411
1412        let identity = SessionIdentity {
1413            source: SessionSource::Claude,
1414            session_id: remote_session.session_id.clone(),
1415        };
1416        clear_pending_suppression_after_outcome(
1417            maintenance_store.as_ref(),
1418            &pending_suppression_clears,
1419            &identity,
1420            if operation == SyncOperation::Unchanged {
1421                SuppressionApplyOutcome::Unchanged
1422            } else {
1423                SuppressionApplyOutcome::Written
1424            },
1425        );
1426
1427        // Track all sessions (including unchanged) in affected conversations
1428        let relative_path_str = relative_path_for_tracking.to_string_lossy().to_string();
1429        match ConversationSummary::new(
1430            remote_session.session_id.clone(),
1431            relative_path_str.clone(),
1432            remote_session.latest_timestamp(),
1433            remote_session.message_count(),
1434            operation,
1435        ) {
1436            Ok(summary) => affected_conversations.push(summary),
1437            Err(e) => log::warn!("Failed to create summary for {}: {}", relative_path_str, e),
1438        }
1439    }
1440
1441    println!("  {} Merged {} sessions", "✓".green(), merged_count);
1442
1443    let protected_entries = incomplete_reasons
1444        .iter()
1445        .filter_map(|(session_id, reason)| {
1446            remote_revisions
1447                .get(session_id)
1448                .map(|(relative, fingerprint)| PullGuardEntry {
1449                    session_id: session_id.clone(),
1450                    remote_relative_path: relative.clone(),
1451                    remote_fingerprint: fingerprint.clone(),
1452                    reason: reason.clone(),
1453                })
1454        })
1455        .collect::<Vec<_>>();
1456    let completed_entries = completed_sessions
1457        .iter()
1458        .filter_map(|session_id| {
1459            remote_revisions
1460                .get(session_id)
1461                .map(|(relative, fingerprint)| {
1462                    (session_id.clone(), relative.clone(), fingerprint.clone())
1463                })
1464        })
1465        .collect::<Vec<_>>();
1466    PullGuardRegistry::reconcile(protected_entries, completed_entries)?;
1467
1468    // ============================================================================
1469    // CREATE AND SAVE OPERATION RECORD
1470    // ============================================================================
1471    let mut operation_record = OperationRecord::new(
1472        OperationType::Pull,
1473        Some(branch_name.clone()),
1474        affected_conversations.clone(),
1475    );
1476
1477    // Attach the snapshot path to the operation record (only if we created one)
1478    operation_record.snapshot_path = snapshot_path;
1479
1480    // Load operation history and add this operation
1481    let mut history = match OperationHistory::load() {
1482        Ok(h) => h,
1483        Err(e) => {
1484            log::warn!("Failed to load operation history: {}", e);
1485            log::info!("Creating new history...");
1486            OperationHistory::default()
1487        }
1488    };
1489
1490    if let Err(e) = history.add_operation(operation_record) {
1491        log::warn!("Failed to save operation to history: {}", e);
1492        log::info!("Pull completed successfully, but history was not updated.");
1493    }
1494
1495    // ============================================================================
1496    // DISPLAY SUMMARY TO USER
1497    // ============================================================================
1498    println!("\n{}", "=== Pull Summary ===".bold().cyan());
1499
1500    // Show operation statistics
1501    let conflict_count = detector.conflict_count();
1502    let stats_msg = format!(
1503        "  {} Added    {} Conflicts    {} Unchanged    {} Deferred",
1504        format!("{added_count}").green(),
1505        format!("{conflict_count}").yellow(),
1506        format!("{unchanged_count}").dimmed(),
1507        format!("{skipped_changed_count}").cyan(),
1508    );
1509    println!("{stats_msg}");
1510    if filter.use_project_name_only && skipped_no_local_match > 0 {
1511        println!(
1512            "  {} Skipped (no local match): {}",
1513            "!".yellow(),
1514            skipped_no_local_match
1515        );
1516    }
1517    if skipped_existing_target > 0 {
1518        println!(
1519            "  {} Skipped (existing local target): {}",
1520            "!".yellow(),
1521            skipped_existing_target
1522        );
1523    }
1524    if skipped_changed_count > 0 {
1525        println!(
1526            "  {} Skipped (local session changed or unsafe baseline): {}",
1527            "!".yellow(),
1528            skipped_changed_count
1529        );
1530    }
1531    println!();
1532
1533    // Group conversations by project (top-level directory)
1534    let mut by_project: HashMap<String, Vec<&ConversationSummary>> = HashMap::new();
1535    for conv in &affected_conversations {
1536        // Skip unchanged conversations in detailed output
1537        if conv.operation == SyncOperation::Unchanged {
1538            continue;
1539        }
1540
1541        let project = conv
1542            .project_path
1543            .split('/')
1544            .next()
1545            .unwrap_or("unknown")
1546            .to_string();
1547        by_project.entry(project).or_default().push(conv);
1548    }
1549
1550    // Display conversations grouped by project
1551    if !by_project.is_empty() {
1552        println!("{}", "Affected Conversations:".bold());
1553
1554        let mut projects: Vec<_> = by_project.keys().collect();
1555        projects.sort();
1556
1557        for project in projects {
1558            let conversations = &by_project[project];
1559            println!("\n  {} {}/", "Project:".bold(), project.cyan());
1560
1561            for conv in conversations.iter().take(MAX_CONVERSATIONS_TO_DISPLAY) {
1562                let operation_str = match conv.operation {
1563                    SyncOperation::Added => "ADD".green(),
1564                    SyncOperation::Modified => "MOD".cyan(),
1565                    SyncOperation::Conflict => "CONFLICT".yellow(),
1566                    SyncOperation::Unchanged => "---".dimmed(),
1567                };
1568
1569                let timestamp_str = conv
1570                    .timestamp
1571                    .as_ref()
1572                    .and_then(|t| {
1573                        // Extract just the date portion for compact display
1574                        t.split('T').next()
1575                    })
1576                    .unwrap_or("unknown");
1577
1578                println!(
1579                    "    {} {} ({}msg, {})",
1580                    operation_str,
1581                    conv.project_path,
1582                    conv.message_count,
1583                    timestamp_str.dimmed()
1584                );
1585            }
1586
1587            if conversations.len() > MAX_CONVERSATIONS_TO_DISPLAY {
1588                println!(
1589                    "    {} ... and {} more conversations",
1590                    "...".dimmed(),
1591                    conversations.len() - MAX_CONVERSATIONS_TO_DISPLAY
1592                );
1593            }
1594        }
1595    }
1596
1597    println!("\n{}", "Pull complete!".green().bold());
1598
1599    // Clean up old snapshots automatically
1600    if let Err(e) = crate::undo::cleanup_old_snapshots(None, false) {
1601        log::warn!("Failed to cleanup old snapshots: {}", e);
1602    }
1603
1604    // ============================================================================
1605    // SYNC AUTO MEMORY DIRECTORIES
1606    // ============================================================================
1607    if filter.auto_memory.enabled {
1608        println!("  {} auto memory directories...", "Syncing".cyan());
1609
1610        let synced_projects = sync_auto_memory_from_remote(
1611            &state.sync_repo_path,
1612            &remote_projects_dir,
1613            &claude_dir,
1614            filter.use_project_name_only,
1615        )?;
1616        if verbosity == VerbosityLevel::Verbose {
1617            for project_name in &synced_projects {
1618                println!("    {} {}/memory", "←".cyan(), project_name);
1619            }
1620        }
1621        let synced_count = synced_projects.len();
1622
1623        if verbosity != VerbosityLevel::Quiet {
1624            println!(
1625                "  {} Synced {} memory directories",
1626                "✓".green(),
1627                synced_count
1628            );
1629        }
1630    }
1631
1632    // Auto-apply CLAUDE.md if enabled
1633    if filter.config_sync.enabled && filter.config_sync.auto_apply_claude_md {
1634        if let Err(e) = crate::handlers::config_sync::auto_apply_claude_md(&filter.config_sync) {
1635            log::debug!("Failed to auto-apply CLAUDE.md: {}", e);
1636        }
1637    }
1638
1639    if !incomplete_reasons.is_empty() {
1640        let mut sessions = incomplete_reasons
1641            .into_iter()
1642            .map(|(session_id, reason)| PullIncompleteSession { session_id, reason })
1643            .collect::<Vec<_>>();
1644        sessions.sort_by(|a, b| a.session_id.cmp(&b.session_id));
1645        return Err(PullIncomplete { sessions }.into());
1646    }
1647
1648    Ok(())
1649}
