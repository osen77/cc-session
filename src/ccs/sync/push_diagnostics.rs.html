1//! Content-free, operation-scoped push diagnostics. No authorization is cached here.
2use std::cell::Cell;
3use std::time::Instant;
4
5#[derive(Clone, Copy, Default)]
6struct Counters {
7    enumerate: u64,
8    probe: u64,
9    diskutil: u64,
10    candidates: u64,
11    parsed: u64,
12    snapshot_bytes: u64,
13    snapshot_us: u64,
14}
15
16thread_local! {
17    static COUNTERS: Cell<Option<Counters>> = const { Cell::new(None) };
18}
19
20pub(crate) struct Operation;
21impl Operation {
22    pub(crate) fn start(enabled: bool) -> Self {
23        COUNTERS.with(|slot| slot.set(enabled.then(Counters::default)));
24        Self
25    }
26}
27impl Drop for Operation {
28    fn drop(&mut self) {
29        COUNTERS.with(|slot| slot.set(None));
30    }
31}
32
33fn count(update: impl FnOnce(&mut Counters)) {
34    COUNTERS.with(|slot| {
35        if let Some(mut counters) = slot.get() {
36            update(&mut counters);
37            slot.set(Some(counters));
38        }
39    });
40}
41pub(crate) fn enumerate() { count(|c| c.enumerate += 1); }
42pub(crate) fn probe() { count(|c| c.probe += 1); }
43pub(crate) fn diskutil() { count(|c| c.diskutil += 1); }
44pub(crate) fn candidate() { count(|c| c.candidates += 1); }
45pub(crate) fn parsed() { count(|c| c.parsed += 1); }
46pub(crate) fn snapshot(bytes: u64, elapsed: std::time::Duration) {
47    count(|c| {
48        c.snapshot_bytes += bytes;
49        c.snapshot_us += elapsed.as_micros().min(u64::MAX as u128) as u64;
50    });
51}
52
53pub(crate) struct Stage {
54    name: &'static str,
55    started: Instant,
56}
57impl Stage {
58    pub(crate) fn start(name: &'static str) -> Self {
59        if COUNTERS.with(|slot| slot.get().is_some()) {
60            eprintln!("CCS_PUSH_PERF event=start stage={name}");
61        }
62        Self { name, started: Instant::now() }
63    }
64}
65impl Drop for Stage {
66    fn drop(&mut self) {
67        COUNTERS.with(|slot| {
68            if let Some(c) = slot.get() {
69                eprintln!("CCS_PUSH_PERF event=end stage={} elapsed_ms={} enumerate={} probe={} diskutil={} candidates={} parsed={} snapshot_bytes={} snapshot_us={}", self.name, self.started.elapsed().as_millis(), c.enumerate, c.probe, c.diskutil, c.candidates, c.parsed, c.snapshot_bytes, c.snapshot_us);
70            }
71        });
72    }
73}
