40/// FTS5 pre-filter result for accelerating `ccs session search` over `agy` sessions.
41#[derive(Debug, Clone, Default)]
42pub struct AgyFtsPrefilter {
43    /// Conversations that have indexed body chunks in `conversations_fts`, mapped to
44    /// their indexed `last_modified_time` in Unix seconds.
45    pub indexed_mtimes: HashMap<String, i64>,
46    /// Conversation IDs matching at least one search keyword in `conversations_fts`.
47    pub matched_ids: HashSet<String>,
48}
49
50impl AgyFtsPrefilter {
51    /// Decide whether a session must be parsed during search.
52    ///
53    /// Returns `false` only when the session is confirmed to be freshly indexed in
54    /// `conversations_fts` (its transcript file is not newer than the FTS index) and
55    /// none of the search keywords matched it in SQLite.
56    pub fn should_scan_session(&self, conversation_id: &str, file_path: &Path) -> bool {
57        if self.matched_ids.contains(conversation_id) {
58            return true;
59        }
60        let Some(&indexed_mtime_secs) = self.indexed_mtimes.get(conversation_id) else {
61            // Not yet indexed in conversations_fts (e.g. brand-new or running session).
62            return true;
63        };
64        if indexed_mtime_secs <= 0 {
65            return true;
66        }
67        // DB-only sessions point at conversation_summaries.db itself; their content is
68        // already represented in the DB.
69        if is_agy_db_path(file_path) {
70            return false;
71        }
72        match fs::metadata(file_path)
73            .ok()
74            .and_then(|m| m.modified().ok())
75            .and_then(|t| t.duration_since(UNIX_EPOCH).ok())
76            .and_then(|d| i64::try_from(d.as_secs()).ok())
77        {
78            // Allow a 2-second slack between DB timestamp and filesystem mtime.
79            Some(file_mtime) => file_mtime > indexed_mtime_secs.saturating_add(2),
80            None => true,
81        }
82    }
83}

137/// Resolve the preferred transcript path for `conversation_id` under `agy_root`.
138///
139/// Returns `(show_path, summary_path)` where `show_path` prefers `transcript_full.jsonl`
140/// (untruncated content for `show` and `search`) and `summary_path` prefers the lighter
141/// `transcript.jsonl` (falling back to whichever file exists).
142pub fn resolve_transcript_paths(
143    agy_root: &Path,
144    conversation_id: &str,
145) -> Option<(PathBuf, PathBuf)> {
146    let logs_dir = agy_brain_dir(agy_root)
147        .join(conversation_id)
148        .join(".system_generated")
149        .join("logs");
150    let full = logs_dir.join("transcript_full.jsonl");
151    let compact = logs_dir.join("transcript.jsonl");
152    let has_full = is_regular_non_symlink(&full);
153    let has_compact = is_regular_non_symlink(&compact);
154    match (has_full, has_compact) {
155        (true, true) => Some((full, compact)),
156        (true, false) => Some((full.clone(), full)),
157        (false, true) => Some((compact.clone(), compact)),
158        (false, false) => None,
159    }
160}

162fn is_regular_non_symlink(path: &Path) -> bool {
163    matches!(
164        fs::symlink_metadata(path),
165        Ok(meta) if !meta.file_type().is_symlink() && meta.is_file()
166    )
167}

169/// Open `conversation_summaries.db` in read-only, WAL-safe mode without creating or locking it.
170///
171/// Returns `Ok(None)` when the database does not exist so callers silently skip `agy` when
172/// Antigravity CLI is not installed.
173pub fn open_readonly_db(db_path: &Path) -> Result<Option<Connection>> {
174    match fs::symlink_metadata(db_path) {
175        Ok(meta) if !meta.file_type().is_symlink() && meta.is_file() => {}
176        Ok(_) => anyhow::bail!("agy database is not a regular file: {}", db_path.display()),
177        Err(err) if err.kind() == std::io::ErrorKind::NotFound => return Ok(None),
178        Err(err) => {
179            return Err(err)
180                .with_context(|| format!("Failed to stat agy database: {}", db_path.display()));
181        }
182    }
183
184    let flags = OpenFlags::SQLITE_OPEN_READ_ONLY | OpenFlags::SQLITE_OPEN_NO_MUTEX;
185    let conn = Connection::open_with_flags(db_path, flags).with_context(|| {
186        format!(
187            "Failed to open agy database read-only: {}",
188            db_path.display()
189        )
190    })?;
191    conn.busy_timeout(Duration::from_millis(2000))?;
192    let _ = conn.execute_batch("PRAGMA query_only = ON;");
193    Ok(Some(conn))
194}

196/// Load conversation summaries from `conversation_summaries.db`.
197///
198/// Returns an empty vector if the database file does not exist.
199pub fn load_db_summaries(db_path: &Path) -> Result<Vec<AgyDbSummary>> {
200    let Some(conn) = open_readonly_db(db_path)? else {
201        return Ok(Vec::new());
202    };
203
204    // Load optional FTS metadata (workspace fallback + archived flag) if table exists.
205    let fts_meta = load_fts_meta_map(&conn);
206
207    let mut stmt = conn.prepare(
208        "SELECT conversation_id, title, preview, step_count, last_modified_time, \
209                workspace_uris, project_id, parent_conversation_id, last_user_input_time \
210         FROM conversation_summaries \
211         ORDER BY last_modified_time DESC",
212    )?;
213
214    let rows = stmt.query_map([], |row| {
215        let conversation_id: String = row.get(0)?;
216        let title: String = row.get::<_, Option<String>>(1)?.unwrap_or_default();
217        let preview: String = row.get::<_, Option<String>>(2)?.unwrap_or_default();
218        let step_count: i64 = row.get::<_, Option<i64>>(3)?.unwrap_or(0);
219        let last_modified_raw: String = row.get::<_, Option<String>>(4)?.unwrap_or_default();
220        let workspace_uris: String = row.get::<_, Option<String>>(5)?.unwrap_or_default();
221        let project_id: String = row.get::<_, Option<String>>(6)?.unwrap_or_default();
222        let parent_conversation_id: String = row.get::<_, Option<String>>(7)?.unwrap_or_default();
223        let last_user_input_raw: String = row.get::<_, Option<String>>(8)?.unwrap_or_default();
224        Ok((
225            conversation_id,
226            title,
227            preview,
228            step_count,
229            last_modified_raw,
230            workspace_uris,
231            project_id,
232            parent_conversation_id,
233            last_user_input_raw,
234        ))
235    })?;
236
237    let mut summaries = Vec::new();
238    for row in rows {
239        let (
240            conversation_id,
241            title,
242            preview,
243            step_count,
244            last_modified_raw,
245            workspace_uris,
246            project_id,
247            parent_conversation_id,
248            last_user_input_raw,
249        ) = row?;
250
251        let (fts_workspace, archived) = fts_meta
252            .get(&conversation_id)
253            .cloned()
254            .unwrap_or((None, false));
255
256        let cwd = parse_first_workspace_uri(&workspace_uris);
257        let project_name =
258            resolve_project_name(cwd.as_deref(), fts_workspace.as_deref(), &project_id);
259
260        summaries.push(AgyDbSummary {
261            conversation_id,
262            title: title.trim().to_string(),
263            preview: preview.trim().to_string(),
264            step_count: usize::try_from(step_count.max(0)).unwrap_or(0),
265            last_modified_time: normalize_agy_timestamp(&last_modified_raw),
266            cwd,
267            project_name,
268            parent_conversation_id: parent_conversation_id.trim().to_string(),
269            last_user_input_time: normalize_agy_timestamp(&last_user_input_raw),
270            archived,
271        });
272    }
273
274    Ok(summaries)
275}

277fn load_fts_meta_map(conn: &Connection) -> HashMap<String, (Option<String>, bool)> {
278    let mut map = HashMap::new();
279    let Ok(mut stmt) = conn.prepare(
280        "SELECT conversation_id, workspace, archived \
281         FROM conversations_fts \
282         WHERE chunk_index = -1",
283    ) else {
284        return map;
285    };
286    let Ok(rows) = stmt.query_map([], |row| {
287        let id: String = row.get(0)?;
288        let workspace: String = row.get::<_, Option<String>>(1)?.unwrap_or_default();
289        let archived: i64 = row.get::<_, Option<i64>>(2)?.unwrap_or(0);
290        Ok((id, workspace, archived != 0))
291    }) else {
292        return map;
293    };
294    for row in rows.flatten() {
295        let (id, ws, archived) = row;
296        let ws = percent_decode(&ws);
297        let ws = ws.trim().to_string();
298        let ws_opt = (!ws.is_empty()).then_some(ws);
299        map.insert(id, (ws_opt, archived));
300    }
301    map
302}

304/// Pre-filter indexed `agy` sessions using SQLite FTS5 (`conversations_fts`).
305///
306/// Uses FTS5 `MATCH` as the fast path for word/phrase queries and supplements with
307/// SQLite substring checks on `conversations_fts` for CJK, punctuation, or infix queries
308/// that `unicode61` cannot match by prefix alone. Returns `None` if the database or
309/// `conversations_fts` table is unavailable (callers fall back to transcript scanning).
310pub fn fts_prefilter_candidates(
311    db_path: &Path,
312    keywords: &[&str],
313    user_only: bool,
314    phrase: bool,
315) -> Option<AgyFtsPrefilter> {
316    if keywords.is_empty() {
317        return None;
318    }
319    let conn = open_readonly_db(db_path).ok()??;
320
321    // Collect every conversation that has at least one indexed body chunk along with
322    // its header row's `last_modified_time` (stored as Unix nanoseconds by agy).
323    let mut indexed_mtimes: HashMap<String, i64> = HashMap::new();
324    {
325        let mut stmt = conn
326            .prepare(
327                "SELECT conversation_id, \
328                        MAX(CASE WHEN chunk_index = -1 THEN last_modified_time ELSE 0 END) AS header_mtime, \
329                        SUM(CASE WHEN chunk_index >= 0 THEN 1 ELSE 0 END) AS body_chunks \
330                 FROM conversations_fts \
331                 GROUP BY conversation_id",
332            )
333            .ok()?;
334        let rows = stmt
335            .query_map([], |row| {
336                let id: String = row.get(0)?;
337                let mtime_raw: i64 = row.get::<_, Option<i64>>(1)?.unwrap_or(0);
338                let body_chunks: i64 = row.get::<_, Option<i64>>(2)?.unwrap_or(0);
339                Ok((id, mtime_raw, body_chunks))
340            })
341            .ok()?;
342        for row in rows.flatten() {
343            let (id, mtime_raw, body_chunks) = row;
344            if body_chunks > 0 {
345                indexed_mtimes.insert(id, fts_timestamp_to_secs(mtime_raw));
346            }
347        }
348    }
349
350    if indexed_mtimes.is_empty() {
351        return Some(AgyFtsPrefilter::default());
352    }
353
354    let mut matched_ids = HashSet::new();
355    let fts_columns = if user_only {
356        "user_body"
357    } else {
358        "{user_body agent_body}"
359    };
360
361    for &kw in keywords {
362        let trimmed = kw.trim();
363        if trimmed.is_empty() {
364            continue;
365        }
366
367        let mut kw_hits = HashSet::new();
368        let can_use_fts_only = is_fts_word_query(trimmed);
369
370        // 1. Primary fast path: SQLite FTS5 inverted index MATCH.
371        if let Some(expr) = build_fts5_match_expr(fts_columns, trimmed, phrase) {
372            if let Ok(mut stmt) = conn.prepare(
373                "SELECT DISTINCT conversation_id \
374                 FROM conversations_fts \
375                 WHERE chunk_index >= 0 AND conversations_fts MATCH ?1",
376            ) {
377                if let Ok(rows) = stmt.query_map([&expr], |row| row.get::<_, String>(0)) {
378                    for id in rows.flatten() {
379                        kw_hits.insert(id);
380                    }
381                }
382            }
383        }
384
385        // 2. Fallback inside SQLite `conversations_fts` for CJK / punctuation / infix
386        //    substrings that `unicode61` prefix indexing does not cover.
387        if !can_use_fts_only || kw_hits.is_empty() {
388            let pattern = format!("%{}%", escape_like_pattern(&trimmed.to_lowercase()));
389            let sql = if user_only {
390                "SELECT DISTINCT conversation_id \
391                 FROM conversations_fts \
392                 WHERE chunk_index >= 0 AND lower(user_body) LIKE ?1 ESCAPE '\\'"
393            } else {
394                "SELECT DISTINCT conversation_id \
395                 FROM conversations_fts \
396                 WHERE chunk_index >= 0 \
397                   AND (lower(user_body) LIKE ?1 ESCAPE '\\' OR lower(agent_body) LIKE ?1 ESCAPE '\\')"
398            };
399            if let Ok(mut stmt) = conn.prepare(sql) {
400                if let Ok(rows) = stmt.query_map([&pattern], |row| row.get::<_, String>(0)) {
401                    for id in rows.flatten() {
402                        kw_hits.insert(id);
403                    }
404                }
405            }
406        }
407
408        matched_ids.extend(kw_hits);
409    }
410
411    Some(AgyFtsPrefilter {
412        indexed_mtimes,
413        matched_ids,
414    })
415}

417/// Convert `conversations_fts.last_modified_time` (nanoseconds, or microseconds/millis/secs)
418/// into Unix seconds.
419fn fts_timestamp_to_secs(raw: i64) -> i64 {
420    if raw <= 0 {
421        return 0;
422    }
423    if raw > 1_000_000_000_000_000_000 {
424        // Nanoseconds (standard Go UnixNano used by agy)
425        raw / 1_000_000_000
426    } else if raw > 1_000_000_000_000_000 {
427        // Microseconds
428        raw / 1_000_000
429    } else if raw > 1_000_000_000_000 {
430        // Milliseconds
431        raw / 1_000
432    } else {
433        raw
434    }
435}

437/// Return true if `kw` consists solely of ASCII alphanumeric characters (and spaces),
438/// which `unicode61` tokenizes predictably.
439fn is_fts_word_query(kw: &str) -> bool {
440    !kw.is_empty()
441        && kw
442            .chars()
443            .all(|c| c.is_ascii_alphanumeric() || c.is_ascii_whitespace())
444}
445
446/// Build a safe FTS5 MATCH expression for a column selector and keyword/phrase.
447fn build_fts5_match_expr(columns: &str, keyword: &str, phrase: bool) -> Option<String> {
448    let trimmed = keyword.trim();
449    if trimmed.is_empty() {
450        return None;
451    }
452    let escaped = trimmed.replace('"', "\"\"");
453    if phrase || trimmed.contains(char::is_whitespace) {
454        Some(format!("{columns} : \"{escaped}\""))
455    } else if trimmed.chars().all(|c| c.is_ascii_alphanumeric()) {
456        Some(format!("{columns} : \"{escaped}\"*"))
457    } else {
458        Some(format!("{columns} : \"{escaped}\""))
459    }
460}
461
462fn escape_like_pattern(input: &str) -> String {
463    let mut out = String::with_capacity(input.len());
464    for ch in input.chars() {
465        match ch {
466            '\\' | '%' | '_' => {
467                out.push('\\');
468                out.push(ch);
469            }
470            _ => out.push(ch),
471        }
472    }
473    out
474}

476/// Load fallback display messages directly from `conversation_summaries.db` when a
477/// conversation has no `transcript*.jsonl` file on disk.
478pub fn load_display_messages_from_db(
479    db_path: &Path,
480    conversation_id: &str,
481) -> Result<Vec<AgyDisplayMessage>> {
482    let Some(conn) = open_readonly_db(db_path)? else {
483        return Ok(Vec::new());
484    };
485
486    let mut messages = Vec::new();
487
488    if let Ok(mut stmt) = conn.prepare(
489        "SELECT user_body, agent_body \
490         FROM conversations_fts \
491         WHERE conversation_id = ?1 AND chunk_index >= 0 \
492         ORDER BY chunk_index ASC",
493    ) {
494        if let Ok(rows) = stmt.query_map([conversation_id], |row| {
495            let user_body: String = row.get::<_, Option<String>>(0)?.unwrap_or_default();
496            let agent_body: String = row.get::<_, Option<String>>(1)?.unwrap_or_default();
497            Ok((user_body, agent_body))
498        }) {
499            for (user_body, agent_body) in rows.flatten() {
500                let u = user_body.trim();
501                if !u.is_empty() {
502                    messages.push(AgyDisplayMessage {
503                        role: "user".to_string(),
504                        timestamp: None,
505                        content: u.to_string(),
506                    });
507                }
508                let a = agent_body.trim();
509                if !a.is_empty() {
510                    messages.push(AgyDisplayMessage {
511                        role: "assistant".to_string(),
512                        timestamp: None,
513                        content: a.to_string(),
514                    });
515                }
516            }
517        }
518    }
519
520    if messages.is_empty() {
521        if let Ok(mut stmt) = conn.prepare(
522            "SELECT preview, last_modified_time \
523             FROM conversation_summaries \
524             WHERE conversation_id = ?1",
525        ) {
526            if let Ok((preview, ts_raw)) = stmt.query_row([conversation_id], |row| {
527                let preview: String = row.get::<_, Option<String>>(0)?.unwrap_or_default();
528                let ts: String = row.get::<_, Option<String>>(1)?.unwrap_or_default();
529                Ok((preview, ts))
530            }) {
531                let text = preview.trim();
532                if !text.is_empty() {
533                    messages.push(AgyDisplayMessage {
534                        role: "user".to_string(),
535                        timestamp: normalize_agy_timestamp(&ts_raw),
536                        content: text.to_string(),
537                    });
538                }
539            }
540        }
541    }
542
543    Ok(messages)
544}

546impl AgySession {
547    /// Parse an `agy` transcript JSONL file.
548    pub fn from_file<P: AsRef<Path>>(path: P) -> Result<Self> {
549        Ok(Self::from_file_with_report(path)?.value)
550    }
551
552    /// Parse an `agy` transcript JSONL file and report any malformed lines skipped.
553    pub fn from_file_with_report<P: AsRef<Path>>(path: P) -> Result<ParseOutcome<Self>> {
554        let path = path.as_ref();
555        let file =
556            File::open(path).with_context(|| format!("Failed to open file: {}", path.display()))?;
557        let reader = BufReader::new(file);
558
559        let mut entries = Vec::new();
560        let mut malformed_lines = 0;
561
562        for (line_num, line) in reader.lines().enumerate() {
563            let line = line.with_context(|| {
564                format!("Failed to read line {} in {}", line_num + 1, path.display())
565            })?;
566            if line.trim().is_empty() {
567                continue;
568            }
569
570            let value: Value = match serde_json::from_str(&line) {
571                Ok(value) => value,
572                Err(e) => {
573                    malformed_lines += 1;
574                    log::debug!(
575                        "Skipping malformed agy transcript line {} in {}: {}",
576                        line_num + 1,
577                        path.display(),
578                        e
579                    );
580                    continue;
581                }
582            };
583
584            let step_index = value.get("step_index").and_then(|v| v.as_u64());
585            let source = value
586                .get("source")
587                .and_then(|v| v.as_str())
588                .unwrap_or_default()
589                .to_string();
590            let entry_type = value
591                .get("type")
592                .and_then(|v| v.as_str())
593                .unwrap_or_default()
594                .to_string();
595            let timestamp = value
596                .get("created_at")
597                .and_then(|v| v.as_str())
598                .and_then(normalize_agy_timestamp);
599            let content = value
600                .get("content")
601                .and_then(|v| v.as_str())
602                .map(|s| s.to_string());
603
604            entries.push(AgyEntry {
605                step_index,
606                source,
607                entry_type,
608                timestamp,
609                content,
610            });
611        }
612
613        let session_id = session_id_from_transcript_path(path)
614            .with_context(|| format!("No agy conversation ID found for: {}", path.display()))?;
615
616        Ok(ParseOutcome {
617            value: Self {
618                session_id,
619                entries,
620                file_path: path.to_path_buf(),
621                cwd: None,
622                title: None,
623            },
624            malformed_lines,
625        })
626    }

628    /// Extract clean user and assistant display messages from this transcript.
629    ///
630    /// - `USER_INPUT`: strips `<USER_REQUEST>`, `<ADDITIONAL_METADATA>`, and
631    ///   `<USER_SETTINGS_CHANGE>` wrappers and filters system injections.
632    /// - `PLANNER_RESPONSE`: extracts non-empty `content` (skipping tool-only steps
633    ///   and internal `thinking`).
634    /// - `GENERIC`, `SYSTEM_MESSAGE`, `ERROR_MESSAGE`: skipped.
635    pub fn display_messages(&self) -> Vec<AgyDisplayMessage> {
636        let mut messages = Vec::new();
637
638        for entry in &self.entries {
639            match entry.entry_type.as_str() {
640                "USER_INPUT" => {
641                    if let Some(raw) = entry.content.as_deref() {
642                        let cleaned = strip_user_request_wrapper(raw);
643                        if !cleaned.is_empty() && !is_system_content(&cleaned) {
644                            messages.push(AgyDisplayMessage {
645                                role: "user".to_string(),
646                                timestamp: entry.timestamp.clone(),
647                                content: cleaned,
648                            });
649                        }
650                    }
651                }
652                "PLANNER_RESPONSE" => {
653                    if let Some(raw) = entry.content.as_deref() {
654                        let trimmed = raw.trim();
655                        if !trimmed.is_empty() && !is_system_content(trimmed) {
656                            messages.push(AgyDisplayMessage {
657                                role: "assistant".to_string(),
658                                timestamp: entry.timestamp.clone(),
659                                content: trimmed.to_string(),
660                            });
661                        }
662                    }
663                }
664                _ => {}
665            }
666        }
667
668        messages
669    }

712/// Strip `<USER_REQUEST>...</USER_REQUEST>`, `<ADDITIONAL_METADATA>...</ADDITIONAL_METADATA>`,
713/// and `<USER_SETTINGS_CHANGE>...</USER_SETTINGS_CHANGE>` wrappers from an `agy` user prompt.
714pub fn strip_user_request_wrapper(raw: &str) -> String {
715    let trimmed = raw.trim();
716    if trimmed.is_empty() {
717        return String::new();
718    }
719
720    if let Some(after_open) = trimmed.strip_prefix("<USER_REQUEST>") {
721        let after_open = after_open
722            .strip_prefix("\r\n")
723            .or_else(|| after_open.strip_prefix('\n'))
724            .unwrap_or(after_open);
725        // Use rfind so a prompt that literally mentions `</USER_REQUEST>` keeps its full body.
726        if let Some(close_idx) = after_open.rfind("</USER_REQUEST>") {
727            return after_open[..close_idx].trim().to_string();
728        }
729        return strip_trailing_metadata_tags(after_open).trim().to_string();
730    }
731
732    strip_trailing_metadata_tags(trimmed).trim().to_string()
733}
734
735fn strip_trailing_metadata_tags(input: &str) -> String {
736    let mut text = input.to_string();
737    for tag in [
738        "ADDITIONAL_METADATA",
739        "USER_SETTINGS_CHANGE",
740        "SYSTEM_MESSAGE",
741    ] {
742        let open = format!("<{tag}>");
743        let close = format!("</{tag}>");
744        loop {
745            let Some(start) = text.find(&open) else {
746                break;
747            };
748            if let Some(end_rel) = text[start + open.len()..].find(&close) {
749                let end = start + open.len() + end_rel + close.len();
750                text.replace_range(start..end, "");
751            } else {
752                // Unclosed trailing metadata block (e.g. truncated transcript line).
753                text.truncate(start);
754                break;
755            }
756        }
757    }
758    text
759}

761/// Normalize an `agy` timestamp (SQLite `'YYYY-MM-DD HH:MM:SS.ffffff+00:00'` or RFC3339)
762/// into canonical UTC RFC3339 (`YYYY-MM-DDTHH:MM:SSZ`). Returns `None` for empty or
763/// Go zero-time (`0001-01-01...`) values.
764pub fn normalize_agy_timestamp(raw: &str) -> Option<String> {
765    let trimmed = raw.trim();
766    if trimmed.is_empty() || trimmed.starts_with("0001-01-01") {
767        return None;
768    }
769    let normalized = if trimmed.len() >= 19 && trimmed.as_bytes().get(10) == Some(&b' ') {
770        let mut s = trimmed.to_string();
771        s.replace_range(10..11, "T");
772        s
773    } else {
774        trimmed.to_string()
775    };
776    DateTime::parse_from_rfc3339(&normalized).ok().map(|dt| {
777        dt.with_timezone(&Utc)
778            .to_rfc3339_opts(SecondsFormat::Secs, true)
779    })
780}

782/// Extract and decode the first workspace path from `conversation_summaries.workspace_uris`.
783pub fn parse_first_workspace_uri(raw: &str) -> Option<String> {
784    let trimmed = raw.trim();
785    if trimmed.is_empty() {
786        return None;
787    }
788    if let Ok(Value::Array(items)) = serde_json::from_str::<Value>(trimmed) {
789        for item in items {
790            if let Some(uri) = item.as_str() {
791                if let Some(path) = decode_workspace_uri(uri) {
792                    return Some(path);
793                }
794            }
795        }
796        return None;
797    }
798    decode_workspace_uri(trimmed)
799}

801/// Decode a `file://` workspace URI into a local filesystem path (supports UTF-8 percent-encoding
802/// and Windows drive-letter URIs like `file:///C:/Users/...`).
803pub fn decode_workspace_uri(uri: &str) -> Option<String> {
804    let trimmed = uri.trim();
805    if trimmed.is_empty() {
806        return None;
807    }
808    let raw_path = trimmed
809        .strip_prefix("file://localhost")
810        .or_else(|| trimmed.strip_prefix("file://"))
811        .unwrap_or(trimmed);
812    let decoded = percent_decode(raw_path);
813    // Strip leading '/' before Windows drive letter: `/C:/Users/...` -> `C:/Users/...`
814    let cleaned = if decoded.len() >= 3
815        && decoded.as_bytes()[0] == b'/'
816        && decoded.as_bytes()[1].is_ascii_alphabetic()
817        && decoded.as_bytes()[2] == b':'
818    {
819        decoded[1..].to_string()
820    } else {
821        decoded
822    };
823    let cleaned = cleaned.trim().to_string();
824    (!cleaned.is_empty()).then_some(cleaned)
825}
826
827fn percent_decode(input: &str) -> String {
828    let bytes = input.as_bytes();
829    let mut out = Vec::with_capacity(bytes.len());
830    let mut i = 0;
831    while i < bytes.len() {
832        if bytes[i] == b'%' && i + 2 < bytes.len() {
833            if let (Some(hi), Some(lo)) = (hex_nibble(bytes[i + 1]), hex_nibble(bytes[i + 2])) {
834                out.push((hi << 4) | lo);
835                i += 3;
836                continue;
837            }
838        }
839        out.push(bytes[i]);
840        i += 1;
841    }
842    String::from_utf8_lossy(&out).into_owned()
843}
844
845fn hex_nibble(b: u8) -> Option<u8> {
846    match b {
847        b'0'..=b'9' => Some(b - b'0'),
848        b'a'..=b'f' => Some(b - b'a' + 10),
849        b'A'..=b'F' => Some(b - b'A' + 10),
850        _ => None,
851    }
852}

854fn resolve_project_name(
855    cwd: Option<&str>,
856    fts_workspace: Option<&str>,
857    project_id: &str,
858) -> String {
859    if let Some(name) = cwd.and_then(last_path_component) {
860        if !name.is_empty() {
861            return name.to_string();
862        }
863    }
864    if let Some(ws) = fts_workspace {
865        if let Some(name) = last_path_component(ws) {
866            if !name.is_empty() {
867                return name.to_string();
868            }
869        }
870    }
871    let pid = project_id.trim();
872    if !pid.is_empty() && pid != "default-cli-project" && pid != "outside-of-project" {
873        return pid.to_string();
874    }
875    "agy".to_string()
876}

878/// Derive a conversation ID from `.../brain/<conversation-id>/.system_generated/logs/transcript*.jsonl`.
879fn session_id_from_transcript_path(path: &Path) -> Option<String> {
880    // Walk up past `logs` and `.system_generated` if present.
881    let mut cur = path.parent()?;
882    if cur.file_name().and_then(|s| s.to_str()) == Some("logs") {
883        cur = cur.parent()?;
884    }
885    if cur.file_name().and_then(|s| s.to_str()) == Some(".system_generated") {
886        cur = cur.parent()?;
887    }
888    if let Some(dir_name) = cur.file_name().and_then(|s| s.to_str()) {
889        if !dir_name.is_empty() && dir_name != "brain" {
890            return Some(dir_name.to_string());
891        }
892    }
893    path.file_stem()
894        .and_then(|s| s.to_str())
895        .map(str::to_string)
896}
