12pub(crate) const CLASSIFIER_VERSION: u32 = 1;
13pub(crate) const DEFAULT_THRESHOLD: u16 = 70;
14/// Weight of a session that ran inside a temporary root. Set so that a temporary
15/// cwd plus one other weak signal still stays below the threshold, while the
16/// usual throwaway shape (few messages, short duration) crosses it.
17const TEMPORARY_CWD_SCORE: u16 = 40;
18/// Weight of an opening message that repeats verbatim in a tight burst.
19const REPEATED_TITLE_SCORE: u16 = 30;
20/// How many verbatim repeats inside the window make a burst.
21const REPEATED_TITLE_BURST_MIN: usize = 3;
22/// Window that separates a person re-running the same probe from a schedule.
23///
24/// Measured against real data: manual re-runs land three copies within minutes,
25/// while the tightest realistic schedule (hourly) needs two hours for three runs.
26/// Only a sub-half-hourly job with a completely fixed prompt would be caught, and
27/// such a job produces enough sessions to be noticed on its own.
28const REPEATED_TITLE_WINDOW_MINUTES: i64 = 60;
29/// Placeholder used when a session has no extractable opening message. It is not a
30/// real title, so it must never group sessions together.
31const MISSING_TITLE_PLACEHOLDER: &str = "(No title)";
32

141/// Classify one session using the conservative, protection-first policy.
142pub(crate) fn classify(
143    candidate: &MaintenanceCandidate,
144    policy: &ClassifierPolicy,
145    now: DateTime<Utc>,
146) -> ClassificationDecision {
147    if candidate.keep {
148        return keep_decision(ReasonCode::KeepProtection);
149    }
150    if candidate.has_custom_title {
151        return keep_decision(ReasonCode::CustomTitleProtection);
152    }
153    if is_recent(candidate.last_activity, now, policy.hide_after_hours) {
154        return keep_decision(ReasonCode::RecentActivityProtection);
155    }
156    if candidate.message_count > 20 {
157        return keep_decision(ReasonCode::LongConversationProtection);
158    }
159    if conversation_duration(candidate).is_some_and(|duration| duration > Duration::hours(2)) {
160        return keep_decision(ReasonCode::LongConversationProtection);
161    }
162
163    if candidate.explicit_test {
164        return ClassificationDecision {
165            classification: Classification::TestCandidate,
166            score: 100,
167            reasons: vec![ReasonCode::ExplicitTestMarker],
168        };
169    }
170
171    let mut score = 0_u16;
172    let mut reasons = Vec::new();
173
174    if is_fixture_session_id(&candidate.identity.session_id) {
175        score = score.saturating_add(60);
176        reasons.push(ReasonCode::FixtureSessionId);
177    }
178
179    let normalized_title = candidate.title.trim().to_lowercase();
180    if is_trivial_probe_title(&normalized_title) {
181        score = score.saturating_add(35);
182        reasons.push(ReasonCode::ExactTestTitle);
183    }
184
185    if ["fixture", "smoke test", "test brief"]
186        .iter()
187        .any(|keyword| normalized_title.contains(keyword))
188    {
189        score = score.saturating_add(25);
190        reasons.push(ReasonCode::AutomatedValidationTitle);
191    }
192
193    if candidate.user_message_count <= 2 {
194        score = score.saturating_add(20);
195        reasons.push(ReasonCode::FewUserMessages);
196    }
197    if candidate.message_count <= 6 {
198        score = score.saturating_add(10);
199        reasons.push(ReasonCode::FewTotalMessages);
200    }
201    if conversation_duration(candidate)
202        .is_some_and(|duration| (Duration::zero()..=Duration::minutes(15)).contains(&duration))
203    {
204        score = score.saturating_add(15);
205        reasons.push(ReasonCode::ShortDuration);
206    }
207
208    if candidate.repeated_title_burst {
209        score = score.saturating_add(REPEATED_TITLE_SCORE);
210        reasons.push(ReasonCode::RepeatedTitleBurst);
211    }
212
213    if is_temporary_cwd(candidate.cwd.as_deref(), &policy.temporary_roots) {
214        score = score.saturating_add(TEMPORARY_CWD_SCORE);
215        if is_fixture_session_id(&candidate.identity.session_id)
216            && candidate.cwd.as_deref().is_some_and(is_fixture_cwd)
217        {
218            reasons.push(ReasonCode::FixtureTemporaryCwd);
219        } else {
220            reasons.push(ReasonCode::TemporaryCwd);
221        }
222    }
223
224    let classification = if score >= policy.threshold {
225        Classification::TestCandidate
226    } else {
227        Classification::Keep
228    };
229    ClassificationDecision {
230        classification,
231        score,
232        reasons,
233    }
234}

236fn keep_decision(reason: ReasonCode) -> ClassificationDecision {
237    ClassificationDecision {
238        classification: Classification::Keep,
239        score: 0,
240        reasons: vec![reason],
241    }

244fn is_recent(
245    last_activity: Option<DateTime<Utc>>,
246    now: DateTime<Utc>,
247    hide_after_hours: u64,
248) -> bool {
249    let Some(last_activity) = last_activity else {
250        return false;
251    };
252    let max_hours = (i64::MAX as u64) / 3_600;
253    let hide_after = Duration::hours(hide_after_hours.min(max_hours) as i64);
254    now.signed_duration_since(last_activity) < hide_after
255}

257fn conversation_duration(candidate: &MaintenanceCandidate) -> Option<Duration> {
258    Some(
259        candidate
260            .last_activity?
261            .signed_duration_since(candidate.first_activity?),
262    )
263}

265fn is_fixture_session_id(session_id: &str) -> bool {
266    static FIXTURE_ID: OnceLock<Regex> = OnceLock::new();
267    FIXTURE_ID
268        .get_or_init(|| {
269            Regex::new(r"^(cc|cx|om)(-cache)?-task[0-9]+$").expect("valid fixture ID regex")
270        })
271        .is_match(session_id)
272}

274/// Flag each session whose opening message repeats verbatim at least
275/// [`REPEATED_TITLE_BURST_MIN`] times inside [`REPEATED_TITLE_WINDOW_MINUTES`].
276///
277/// A person debugging re-runs the same prompt back to back; a scheduled job repeats
278/// on a cadence measured in hours or days. The window is what separates them, so
279/// grouping is exact and a job whose prompt embeds per-run content never groups at all.
280///
281/// Input is `(title, first_activity)` per session, in any order. Output is a
282/// same-length, same-order flag vector. Sessions without a timestamp or with the
283/// missing-title placeholder are never flagged.
284pub(crate) fn repeated_title_bursts(sessions: &[(String, Option<DateTime<Utc>>)]) -> Vec<bool> {
285    let mut by_title: HashMap<&str, Vec<(DateTime<Utc>, usize)>> = HashMap::new();
286    for (index, (title, first_activity)) in sessions.iter().enumerate() {
287        let trimmed = title.trim();
288        if trimmed.is_empty() || trimmed == MISSING_TITLE_PLACEHOLDER {
289            continue;
290        }
291        if let Some(started_at) = first_activity {
292            by_title
293                .entry(trimmed)
294                .or_default()
295                .push((*started_at, index));
296        }
297    }
298
299    let window = Duration::minutes(REPEATED_TITLE_WINDOW_MINUTES);
300    let mut flags = vec![false; sessions.len()];
301    for occurrences in by_title.values_mut() {
302        if occurrences.len() < REPEATED_TITLE_BURST_MIN {
303            continue;
304        }
305        occurrences.sort_by_key(|(started_at, _)| *started_at);
306        let mut start = 0usize;
307        for end in 0..occurrences.len() {
308            while occurrences[end]
309                .0
310                .signed_duration_since(occurrences[start].0)
311                > window
312            {
313                start += 1;
314            }
315            if end - start + 1 >= REPEATED_TITLE_BURST_MIN {
316                for (_, index) in &occurrences[start..=end] {
317                    flags[*index] = true;
318                }
319            }
320        }
321    }
322    flags
323}

325/// Opening messages that carry no task on their own, so the whole session is a probe.
326///
327/// Matching is exact after trimming trailing punctuation. Substring matching is
328/// deliberately avoided: short genuine questions ("茅台现在多少钱？") share the same
329/// message-count and duration shape as a probe, so the title is the only signal
330/// separating them and it has to be precise.
331const TRIVIAL_PROBE_TITLES: [&str; 11] = [
332    "测试",
333    "test",
334    "hello",
335    "hi",
336    "试一下",
337    "ok",
338    "好",
339    "你好",
340    "在吗",
341    "说一句话",
342    "随便说点什么",
343];
344
345fn is_trivial_probe_title(normalized_title: &str) -> bool {
346    let trimmed = normalized_title.trim_end_matches(|character: char| {
347        character.is_whitespace()
348            || matches!(
349                character,
350                '。' | '．' | '.' | '！' | '!' | '？' | '?' | '~' | '～' | '、' | '，' | ','
351            )
352    });
353    TRIVIAL_PROBE_TITLES.contains(&trimmed)
354}

356fn is_temporary_cwd(cwd: Option<&Path>, temporary_roots: &[PathBuf]) -> bool {
357    let Some(cwd) = cwd else {
358        return false;
359    };
360    temporary_roots
361        .iter()
362        .any(|root| !root.as_os_str().is_empty() && cwd.starts_with(root))
363}
364
365fn is_fixture_cwd(cwd: &Path) -> bool {
366    let Some(name) = cwd.file_name().and_then(|name| name.to_str()) else {
367        return false;
368    };
369    let Some(task_prefix) = name.strip_prefix("task") else {
370        return false;
371    };
372    let Some(task_number) = task_prefix.strip_suffix("-project") else {
373        return false;
374    };
375    !task_number.is_empty()
376        && task_number
377            .chars()
378            .all(|character| character.is_ascii_digit())
379}
