1//! Inter-process mutual exclusion for the sync repository.
2//!
3//! Several entry points write to the same repository — the Stop hook's push,
4//! the SessionStart hook's pull, the launcher wrapper, manual commands and
5//! session deletion. Git does not tolerate concurrent writers: the loser hits
6//! `index.lock` or a non-fast-forward and exits non-zero, which surfaces to the
7//! user as a hook failure.
8//!
9//! Every writer takes this lock first, so they serialize instead of colliding.
10//! The lock is advisory and process-wide, held for the duration of the guard.
11
12use anyhow::Result;
13use colored::Colorize;
14use std::path::Path;
15use std::sync::atomic::{AtomicBool, Ordering};
16use std::time::Duration;
17
18use crate::atomic_file::FileLock;
19use crate::config::ConfigManager;
20use crate::VerbosityLevel;

22/// How long an interactive invocation waits for a busy repository.
23///
24/// Long enough to outlast a normal push (~20s), short enough that a wedged
25/// process does not hang a terminal indefinitely.
26const INTERACTIVE_WAIT: Duration = Duration::from_secs(120);
27
28/// Whether *this process* already holds the repository lock.
29///
30/// `flock` is per file descriptor, not per process: a second acquisition from
31/// the same process blocks against itself. Nesting two locked calls would
32/// therefore deadlock, and such a deadlock is painful to diagnose. The flag
33/// makes a nested acquisition a no-op instead.
34static REPO_LOCK_HELD: AtomicBool = AtomicBool::new(false);
35
36/// Result of trying to take the repository lock.
37pub(crate) enum RepoLockOutcome {
38    /// The caller now owns exclusive access.
39    Acquired(RepoLock),
40    /// Another process is mid-sync; the caller should skip its work.
41    Busy,
42}
43
44/// RAII guard for exclusive access to one sync repository.
45pub(crate) struct RepoLock {
46    /// `None` for a re-entrant guard, which owns no file lock.
47    _file: Option<FileLock>,
48    /// Only the outermost guard clears the process-wide flag.
49    owns_process_flag: bool,
50}
51
52impl Drop for RepoLock {
53    fn drop(&mut self) {
54        if self.owns_process_flag {
55            REPO_LOCK_HELD.store(false, Ordering::SeqCst);
56        }
57    }
58}
59
60impl RepoLock {
61    /// Take the lock, waiting only when a human is watching.
62    ///
63    /// Non-interactive callers (hooks, wrappers, scripts) must not stall: the
64    /// Claude Code hook harness kills them after 60s and reports an error, so
65    /// waiting merely trades one failure for another. They get `Busy`
66    /// immediately. Interactive callers typed the command and expect it to run,
67    /// so they poll up to [`INTERACTIVE_WAIT`].
68    pub(crate) fn acquire(repo_path: &Path) -> Result<RepoLockOutcome> {
69        // Re-entrant: this process already holds it, so the caller is nested
70        // inside another locked section and is already serialized.
71        if REPO_LOCK_HELD
72            .compare_exchange(false, true, Ordering::SeqCst, Ordering::SeqCst)
73            .is_err()
74        {
75            return Ok(RepoLockOutcome::Acquired(Self {
76                _file: None,
77                owns_process_flag: false,
78            }));
79        }
80
81        // From here on the flag is set, so every exit path must clear it unless
82        // it hands back an owning guard.
83        let result = Self::acquire_file_lock(repo_path);
84        match result {
85            Ok(Some(file)) => Ok(RepoLockOutcome::Acquired(Self {
86                _file: Some(file),
87                owns_process_flag: true,
88            })),
89            Ok(None) => {
90                REPO_LOCK_HELD.store(false, Ordering::SeqCst);
91                Ok(RepoLockOutcome::Busy)
92            }
93            Err(error) => {
94                REPO_LOCK_HELD.store(false, Ordering::SeqCst);
95                Err(error)
96            }
97        }
98    }
99
100    fn acquire_file_lock(repo_path: &Path) -> Result<Option<FileLock>> {
101        let lock_path = ConfigManager::sync_repo_lock_path(repo_path)?;
102
103        if crate::interactive_conflict::is_interactive() {
104            FileLock::acquire_with_timeout(&lock_path, INTERACTIVE_WAIT, || {
105                println!("  {} 另一个同步正在进行，等待中...", "⏳".yellow());
106            })
107        } else {
108            FileLock::try_acquire(&lock_path)
109        }
110    }
111
112    /// Take the lock, reporting contention for the caller.
113    ///
114    /// Returns `None` when the repository is busy — the caller should return
115    /// success without doing work. A skipped sync is not an error: returning
116    /// non-zero here is exactly what makes Claude Code report a hook failure,
117    /// and the next hook will sync anyway.
118    pub(crate) fn acquire_or_report(
119        repo_path: &Path,
120        operation: &str,
121        verbosity: VerbosityLevel,
122    ) -> Result<Option<Self>> {
123        match Self::acquire(repo_path)? {
124            RepoLockOutcome::Acquired(lock) => Ok(Some(lock)),
125            RepoLockOutcome::Busy => {
126                log::info!("another sync is in progress, skipping {operation}");
127                if verbosity != VerbosityLevel::Quiet {
128                    println!(
129                        "  {} 另一个同步正在进行，已跳过本次{operation}",
130                        "ℹ".yellow()
131                    );
132                }
133                Ok(None)
134            }
135        }
136    }
137}

139#[cfg(test)]
140mod tests {
141    use super::*;
142    use serial_test::serial;
143    use tempfile::tempdir;
144
145    /// The process-wide flag is global state; tests that touch it must not run
146    /// concurrently with each other.
147    fn reset_flag() {
148        REPO_LOCK_HELD.store(false, Ordering::SeqCst);
149    }
150
151    #[test]
152    #[serial]
153    fn nested_acquisition_does_not_deadlock() {
154        reset_flag();
155        let dir = tempdir().unwrap();
156        std::env::set_var("CLAUDE_CODE_SYNC_CONFIG_DIR", dir.path());
157
158        let outer = RepoLock::acquire(dir.path()).unwrap();
159        assert!(matches!(outer, RepoLockOutcome::Acquired(_)));
160
161        // Would block forever against its own flock without the re-entrancy
162        // guard, so reaching the assertion at all is the real check.
163        let inner = RepoLock::acquire(dir.path()).unwrap();
164        assert!(matches!(inner, RepoLockOutcome::Acquired(_)));
165
166        drop(inner);
167        // The inner guard must not have released the process flag.
168        assert!(REPO_LOCK_HELD.load(Ordering::SeqCst));
169        drop(outer);
170        assert!(!REPO_LOCK_HELD.load(Ordering::SeqCst));
171
172        std::env::remove_var("CLAUDE_CODE_SYNC_CONFIG_DIR");
173    }
174
175    #[test]
176    #[serial]
177    fn flag_is_cleared_when_guard_drops() {
178        reset_flag();
179        let dir = tempdir().unwrap();
180        std::env::set_var("CLAUDE_CODE_SYNC_CONFIG_DIR", dir.path());
181
182        {
183            let _lock = RepoLock::acquire(dir.path()).unwrap();
184            assert!(REPO_LOCK_HELD.load(Ordering::SeqCst));
185        }
186        assert!(!REPO_LOCK_HELD.load(Ordering::SeqCst));
187
188        // A fresh acquisition must still succeed after the flag is cleared.
189        let again = RepoLock::acquire(dir.path()).unwrap();
190        assert!(matches!(again, RepoLockOutcome::Acquired(_)));
191
192        std::env::remove_var("CLAUDE_CODE_SYNC_CONFIG_DIR");
193    }
194
195    #[test]
196    #[serial]
197    fn lock_path_differs_per_repository() {
198        let config = tempdir().unwrap();
199        std::env::set_var("CLAUDE_CODE_SYNC_CONFIG_DIR", config.path());
200        let first = tempdir().unwrap();
201        let second = tempdir().unwrap();
202
203        let a = ConfigManager::sync_repo_lock_path(first.path()).unwrap();
204        let b = ConfigManager::sync_repo_lock_path(second.path()).unwrap();
205        assert_ne!(a, b, "distinct repositories must not share a lock");
206
207        // Same repository, trailing separator: must map to the same lock.
208        let with_slash = ConfigManager::sync_repo_lock_path(&first.path().join("")).unwrap();
209        assert_eq!(a, with_slash);
210
211        std::env::remove_var("CLAUDE_CODE_SYNC_CONFIG_DIR");
212    }
