16/// Show sync status
17pub fn show_status(show_conflicts: bool, show_files: bool) -> Result<()> {
18    let state = SyncState::load()?;
19    let repo = scm::open(&state.sync_repo_path)?;
20    let filter = FilterConfig::load()?;
21    let root_mappings = filter.root_mappings()?;
22    let claude_dir = claude_projects_dir()?;
23    let local_roots = crate::project_roots::enumerate(&claude_dir, &root_mappings)?;
24
25    println!("{}", "=== Claude Code Sync Status ===".bold().cyan());
26    println!();
27
28    // Installation info
29    println!("{}", "安装信息:".bold());
30    if let Ok(exe_path) = std::env::current_exe() {
31        println!("  二进制: {}", exe_path.display().to_string().dimmed());
32    }
33    if let Ok(config_dir) = ConfigManager::config_dir() {
34        println!("  配置目录: {}", config_dir.display().to_string().dimmed());
35    }
36    println!();
37
38    // Claude Code info
39    println!("{}", "Claude Code:".bold());
40    if let Some(parent) = claude_dir.parent() {
41        println!("  目录: {}", parent.display().to_string().dimmed());
42    }
43    println!();
44
45    // Repository info
46    println!("{}", "同步仓库:".bold());
47    println!("  本地路径: {}", state.sync_repo_path.display());
48    let backend = scm::detect_backend(&state.sync_repo_path)
49        .map(|b| format!("{:?}", b))
50        .unwrap_or_else(|| "Unknown".to_string());
51    println!("  后端: {}", backend);
52
53    // Show remote URL if configured
54    if state.has_remote {
55        if let Ok(remote_url) = repo.get_remote_url("origin") {
56            println!("  远程仓库: {}", remote_url.cyan());
57        } else {
58            println!("  远程仓库: {}", "已配置".green());
59        }
60    } else {
61        println!("  远程仓库: {}", "未配置".yellow());
62    }
63
64    if let Ok(branch) = repo.current_branch() {
65        println!("  分支: {}", branch.cyan());
66    }
67
68    if let Ok(has_changes) = repo.has_changes() {
69        println!(
70            "  未提交变更: {}",
71            if has_changes {
72                "是".yellow()
73            } else {
74                "否".green()
75            }
76        );
77    }
78
79    // Session counts. These are display-only, so use the streaming counter
80    // instead of fully parsing every conversation file.
81    println!();
82    println!("{}", "对话历史:".bold());
83    let local_session_count = count_unique_sessions_in_roots(&local_roots, &filter)?;
84    println!("  本地: {} 个会话", local_session_count.to_string().cyan());
85
86    let remote_projects_dir = state.sync_repo_path.join(&filter.sync_subdirectory);
87    if remote_projects_dir.exists() {
88        validate_sync_projects_root(&state.sync_repo_path, &remote_projects_dir)?;
89        let remote_session_count = count_unique_sessions(&remote_projects_dir, &filter)?;
90        println!(
91            "  同步仓库: {} 个会话",
92            remote_session_count.to_string().cyan()
93        );
94    }
95
96    // Config sync info
97    println!();
98    println!("{}", "配置同步:".bold());
99    let config_sync = &filter.config_sync;
100    println!(
101        "  状态: {}",
102        if config_sync.enabled {
103            "已启用".green()
104        } else {
105            "已禁用".yellow()
106        }
107    );
108    println!("  设备名: {}", config_sync.get_device_name().cyan());
109
110    // Show what's being synced
111    let mut sync_items = Vec::new();
112    if config_sync.sync_settings {
113        sync_items.push("settings.json");
114    }
115    if config_sync.sync_claude_md {
116        sync_items.push("CLAUDE.md");
117    }
118    if config_sync.sync_skills_list {
119        sync_items.push("skills");
120    }
121    if config_sync.sync_hooks {
122        sync_items.push("hooks");
123    }
124    if !sync_items.is_empty() {
125        println!("  同步项: {}", sync_items.join(", "));
126    }
127    println!(
128        "  自动应用 CLAUDE.md: {}",
129        if config_sync.auto_apply_claude_md {
130            "是".green()
131        } else {
132            "否".dimmed()
133        }
134    );
135
136    // Check for configs directory
137    let configs_dir = state.sync_repo_path.join("_configs");
138    if configs_dir.exists() {
139        if let Ok(entries) = std::fs::read_dir(&configs_dir) {
140            let devices: Vec<String> = entries
141                .filter_map(|e| e.ok())
142                .filter(|e| e.path().is_dir())
143                .filter_map(|e| e.file_name().into_string().ok())
144                .collect();
145            if !devices.is_empty() {
146                println!("  可用设备: {}", devices.join(", ").dimmed());
147            }
148        }
149    }
150
151    // Show files if requested; only this detail view needs the full parse.
152    if show_files {
153        println!();
154        println!("{}", "本地会话文件:".bold());
155        let local_sessions = discover_local_sessions(&claude_dir, &filter, false)?;
156        for session in local_sessions.iter().take(20) {
157            let relative = Path::new(&session.file_path)
158                .strip_prefix(&claude_dir)
159                .unwrap_or(Path::new(&session.file_path));
160            println!(
161                "  {} ({} 条消息)",
162                relative.display(),
163                session.message_count()
164            );
165        }
166        if local_sessions.len() > 20 {
167            println!("  ... 还有 {} 个", local_sessions.len() - 20);
168        }
169    }
170
171    // Show conflicts if requested
172    if show_conflicts {
173        println!();
174        if let Ok(report) = crate::report::load_latest_report() {
175            if report.total_conflicts > 0 {
176                report.print_summary();
177            } else {
178                println!("{}", "上次同步无冲突".green());
179            }
180        }
181    }
182
183    Ok(())
184}

197    struct EnvGuard {
198        home: Option<std::ffi::OsString>,
199        config: Option<std::ffi::OsString>,
200    }
201
202    impl Drop for EnvGuard {
203        fn drop(&mut self) {
204            match self.home.take() {
205                Some(value) => std::env::set_var("HOME", value),
206                None => std::env::remove_var("HOME"),
207            }
208            match self.config.take() {
209                Some(value) => std::env::set_var(CONFIG_DIR_ENV, value),
210                None => std::env::remove_var(CONFIG_DIR_ENV),
211            }
212        }
213    }
214
215    #[test]
216    #[serial]
217    fn external_root_status_is_readable_and_fail_closed_without_state_changes() {
218        let temp = tempdir().unwrap();
219        let home = temp.path().join("home");
220        let config_dir = temp.path().join("config");
221        let logical = home.join(".claude/projects");
222        let mount = temp.path().join("volume");
223        let trusted = mount.join("Claude");
224        let target = trusted.join("projects");
225        let project = target.join("project");
226        let repo_path = temp.path().join("repo");
227        fs::create_dir_all(logical.parent().unwrap()).unwrap();
228        fs::create_dir_all(project.join("memory")).unwrap();
229        fs::create_dir_all(&config_dir).unwrap();
230        fs::write(
231            project.join("session.jsonl"),
232            br#"{"type":"user","sessionId":"sid","cwd":"/work/project","message":{"role":"user","content":"hello"}}
233"#,
234        )
235        .unwrap();
236        symlink(&target, &logical).unwrap();
237        crate::scm::init(&repo_path).unwrap();
238
239        let guard = EnvGuard {
240            home: std::env::var_os("HOME"),
241            config: std::env::var_os(CONFIG_DIR_ENV),
242        };
243        std::env::set_var("HOME", &home);
244        std::env::set_var(CONFIG_DIR_ENV, &config_dir);
245        let state = SyncState {
246            sync_repo_path: repo_path,
247            has_remote: false,
248            is_cloned_repo: false,
249            last_synced_commit: None,
250        };
251        state.save().unwrap();
252        let external = ExternalProjectsRoot {
253            target: target.clone(),
254            trusted_root: trusted,
255            volume_uuid: "AF9C9871-18AE-40C9-8D18-624E415520C6".into(),
256        };
257        let mut filter = FilterConfig::default();
258        filter.external_projects_root = Some(external.clone());
259        filter.save().unwrap();
260        let state_before = fs::read(ConfigManager::state_file_path().unwrap()).unwrap();
261        let config_before = fs::read(ConfigManager::filter_config_path().unwrap()).unwrap();
262        let volume = VolumeIdentity {
263            mount_point: mount,
264            volume_uuid: external.volume_uuid.clone(),
265            device_id: {
266                use std::os::unix::fs::MetadataExt;
267                fs::metadata(temp.path()).unwrap().dev()
268            },
269        };
270
271        with_test_volume(volume, || {
272            show_status(false, false).unwrap();
273            fs::remove_dir_all(&target).unwrap();
274            assert!(show_status(false, false).is_err());
275            assert_eq!(fs::read(ConfigManager::state_file_path().unwrap()).unwrap(), state_before);
276            assert_eq!(fs::read(ConfigManager::filter_config_path().unwrap()).unwrap(), config_before);
277        });
278        drop(guard);
279    }
