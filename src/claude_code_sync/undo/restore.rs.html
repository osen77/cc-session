7impl Snapshot {
8    /// Restore files from this snapshot
9    ///
10    /// This writes all files from the snapshot back to their original locations,
11    /// overwriting any current content.
12    ///
13    /// For differential snapshots, this recursively loads the base snapshot chain
14    /// and applies all changes in order to reconstruct the full state.
15    ///
16    /// # Security
17    /// This method validates all paths to prevent path traversal attacks.
18    /// By default, only paths within the home directory are allowed.
19    /// For testing, you can pass a custom allowed_base_dir.
20    ///
21    /// # Arguments
22    /// * `allowed_base_dir` - Optional base directory for path validation.
23    ///   If None, defaults to home directory for security.
24    /// * `snapshots_dir` - Optional snapshots directory (for testing with differential snapshots)
25    pub fn restore_with_base_and_snapshots(
26        &self,
27        allowed_base_dir: Option<&Path>,
28        snapshots_dir: Option<&Path>,
29    ) -> Result<()> {
30        // Determine the allowed base directory
31        let allowed_base = if let Some(base) = allowed_base_dir {
32            // For testing: use the provided base
33            base.canonicalize().with_context(|| {
34                format!("Failed to canonicalize base directory: {}", base.display())
35            })?
36        } else {
37            // For production: use home directory
38            let home_dir = dirs::home_dir().context("Failed to get home directory")?;
39            home_dir
40                .canonicalize()
41                .context("Failed to canonicalize home directory")?
42        };
43        let logical_base = match allowed_base_dir {
44            Some(base) => base.to_path_buf(),
45            None => dirs::home_dir().context("Failed to get home directory")?,
46        };
47
48        // Build the complete file state by walking the snapshot chain
49        let all_files = self.reconstruct_full_state_with_dir(snapshots_dir)?;
50
51        // Validate the complete batch before creating, deleting, or writing anything.
52        // Unknown/mapped symlinks stay forbidden for undo in this release.
53        let relative = |path: &Path| -> Result<PathBuf> {
54            let rel = path
55                .strip_prefix(&logical_base)
56                .or_else(|_| path.strip_prefix(&allowed_base))
57                .map_err(|_| anyhow!("restore path is outside allowed root: {}", path.display()))?;
58            crate::path_security::safe_join_within_root(&allowed_base, rel)?;
59            Ok(rel.to_path_buf())
60        };
61        for path in all_files.keys().chain(self.deleted_files.iter()) {
62            let path = Path::new(path);
63            relative(path)?;
64            match fs::symlink_metadata(path) {
65                Ok(metadata) if !metadata.is_file() || metadata.file_type().is_symlink() => {
66                    return Err(anyhow!(
67                        "restore target is not a regular file: {}",
68                        path.display()
69                    ));
70                }
71                Err(error) if error.kind() != std::io::ErrorKind::NotFound => {
72                    return Err(error.into())
73                }
74                _ => {}
75            }
76        }
77        for deleted_path in &self.deleted_files {
78            let rel = relative(Path::new(deleted_path))?;
79            let path = crate::path_security::safe_join_within_root(&allowed_base, &rel)?;
80            match fs::remove_file(&path) {
81                Ok(()) => {}
82                Err(error) if error.kind() == std::io::ErrorKind::NotFound => {}
83                Err(error) => return Err(error.into()),
84            }
85        }
86        for (path, content) in &all_files {
87            let rel = relative(Path::new(path))?;
88            let destination =
89                crate::path_security::prepare_regular_file_destination(&allowed_base, &rel)?;
90            fs::write(&destination, content)
91                .with_context(|| format!("Failed to restore file: {}", destination.display()))?;
92        }
93
94        Ok(())
95    }
96
97    /// Restore files from this snapshot using default snapshots directory
98    ///
99    /// This is a wrapper around `restore_with_base_and_snapshots` for backwards compatibility.
100    pub fn restore_with_base(&self, allowed_base_dir: Option<&Path>) -> Result<()> {
101        self.restore_with_base_and_snapshots(allowed_base_dir, None)
102    }
103
104    /// Restore files from this snapshot
105    ///
106    /// This is a convenience wrapper that uses the home directory as the allowed base.
107    #[allow(dead_code)]
108    pub fn restore(&self) -> Result<()> {
109        self.restore_with_base(None)
110    }
111}

113#[cfg(test)]
114mod tests {
115    use super::*;
116
117    #[test]
118    fn invalid_restore_destination_does_not_create_paths_or_delete_valid_files() {
119        let root = tempfile::tempdir().unwrap();
120        let outside = tempfile::tempdir().unwrap();
121        let victim = root.path().join("keep.txt");
122        fs::write(&victim, b"keep").unwrap();
123        let forbidden = outside.path().join("must-not-exist/file");
124        let snapshot = Snapshot {
125            snapshot_id: "isolated".into(),
126            timestamp: chrono::Utc::now(),
127            operation_type: crate::history::OperationType::Pull,
128            git_commit_hash: None,
129            files: [(forbidden.to_string_lossy().into_owned(), b"bad".to_vec())].into(),
130            branch: None,
131            base_snapshot_id: None,
132            deleted_files: vec![victim.to_string_lossy().into_owned()],
133        };
134        assert!(snapshot.restore_with_base(Some(root.path())).is_err());
135        assert!(!forbidden.parent().unwrap().exists());
136        assert_eq!(fs::read(victim).unwrap(), b"keep");
137    }
