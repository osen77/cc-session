1//! SCM (Source Control Management) abstraction layer.
2//!
3//! Provides a unified interface for Git and Mercurial using CLI commands.
4//! Backend selection is controlled via the `Backend` enum.

10use anyhow::{anyhow, Error, Result};

13/// Classified git push failures.
14#[derive(Debug)]
15pub enum PushError {
16    /// Remote rejected the push because local history is behind.
17    NonFastForward,
18    /// Any other push failure with source context preserved.
19    Other(Error),
20}
21
22/// Rebase command outcome.
23#[derive(Debug, Clone, Copy, PartialEq, Eq)]
24pub enum RebaseOutcome {
25    /// Rebase completed successfully.
26    Completed,
27    /// Rebase still requires user or programmatic resolution.
28    InProgress,
29}

31pub use git::GitScm;
32pub use hg::HgScm;

34/// SCM backend types.
35///
36/// Used for parameterized testing and explicit backend selection.
37#[derive(Debug, Clone, Copy, PartialEq, Eq)]
38pub enum Backend {
39    /// Git version control
40    Git,
41    /// Mercurial version control
42    Mercurial,
43}
44
45#[allow(dead_code)]
46impl Backend {
47    /// Check if this backend's binary is available on the system.
48    pub fn is_available(&self) -> bool {
49        let binary = match self {
50            Backend::Git => "git",
51            Backend::Mercurial => "hg",
52        };
53        std::process::Command::new(binary)
54            .arg("--version")
55            .output()
56            .map(|o| o.status.success())
57            .unwrap_or(false)
58    }
59
60    /// Get the marker directory for this backend (.git, .hg, etc).
61    pub fn marker(&self) -> &'static str {
62        match self {
63            Backend::Git => ".git",
64            Backend::Mercurial => ".hg",
65        }
66    }
67}

69/// Trait for source control management operations.
70pub trait Scm: Send + Sync {
71    /// Get the current branch name.
72    fn current_branch(&self) -> Result<String>;
73
74    /// Get the current commit hash.
75    fn current_commit_hash(&self) -> Result<String>;
76
77    /// Stage all changes (add and remove).
78    fn stage_all(&self) -> Result<()>;
79
80    /// Commit staged changes with a message.
81    fn commit(&self, message: &str) -> Result<()>;
82
83    /// Check if there are uncommitted changes.
84    fn has_changes(&self) -> Result<bool>;
85
86    /// Add a remote repository.
87    fn add_remote(&self, name: &str, url: &str) -> Result<()>;
88
89    /// Check if a remote exists.
90    fn has_remote(&self, name: &str) -> bool;
91
92    /// Get the URL for a remote.
93    fn get_remote_url(&self, name: &str) -> Result<String>;
94
95    /// Set or update the URL for a remote.
96    fn set_remote_url(&self, name: &str, url: &str) -> Result<()>;
97
98    /// Remove a remote.
99    fn remove_remote(&self, name: &str) -> Result<()>;
100
101    /// List all remote names.
102    fn list_remotes(&self) -> Result<Vec<String>>;
103
104    /// Push to a remote repository.
105    fn push(&self, remote: &str, branch: &str) -> Result<()>;
106
107    /// Push to a remote repository and classify recoverable failures.
108    fn push_classified(&self, remote: &str, branch: &str) -> std::result::Result<(), PushError> {
109        self.push(remote, branch).map_err(PushError::Other)
110    }
111
112    /// Fetch remote changes without updating the working tree.
113    fn fetch(&self, _remote: &str) -> Result<()> {
114        Err(anyhow!("fetch is not supported by this SCM backend"))
115    }
116
117    /// Rebase onto an upstream reference.
118    fn rebase(&self, _upstream: &str) -> Result<RebaseOutcome> {
119        Err(anyhow!("rebase is not supported by this SCM backend"))
120    }
121
122    /// Continue an in-progress rebase.
123    #[allow(dead_code)]
124    fn rebase_continue(&self) -> Result<RebaseOutcome> {
125        Err(anyhow!(
126            "rebase continue is not supported by this SCM backend"
127        ))
128    }
129
130    /// Abort an in-progress rebase.
131    fn rebase_abort(&self) -> Result<()> {
132        Err(anyhow!("rebase abort is not supported by this SCM backend"))
133    }
134
135    /// Check whether a rebase is currently in progress.
136    fn is_rebase_in_progress(&self) -> Result<bool> {
137        Ok(false)
138    }
139
140    /// Pull from a remote repository (fetch + merge/update).
141    fn pull(&self, remote: &str, branch: &str) -> Result<()>;
142
143    /// Reset to a specific commit (soft reset - keeps working directory).
144    fn reset_soft(&self, commit: &str) -> Result<()>;
145}

147/// Check if a directory is a repository (Git or Mercurial).
148pub fn is_repo(path: &Path) -> bool {
149    path.join(".git").exists() || path.join(".hg").exists()
150}
151
152/// Open an existing repository (Git or Mercurial).
153///
154/// Automatically detects the backend based on the marker directory.
155pub fn open(path: &Path) -> Result<Box<dyn Scm>> {
156    if path.join(".git").exists() {
157        Ok(Box::new(GitScm::open(path)?))
158    } else if path.join(".hg").exists() {
159        Ok(Box::new(HgScm::open(path)?))
160    } else {
161        Err(anyhow!(
162            "No repository found at '{}'. Expected .git or .hg directory.",
163            path.display()
164        ))
165    }
166}
167
168/// Initialize a new Git repository.
169pub fn init(path: &Path) -> Result<Box<dyn Scm>> {
170    Ok(Box::new(GitScm::init(path)?))
171}
172
173/// Clone a repository from a URL.
174pub fn clone(url: &str, path: &Path) -> Result<Box<dyn Scm>> {
175    Ok(Box::new(GitScm::clone(url, path)?))
176}
177
178/// Initialize a new repository with the specified backend.
179///
180/// This is useful for parameterized testing where you want to test
181/// the same operations against different SCM backends.
182#[allow(dead_code)]
183pub fn init_with_backend(path: &Path, backend: Backend) -> Result<Box<dyn Scm>> {
184    match backend {
185        Backend::Git => Ok(Box::new(GitScm::init(path)?)),
186        Backend::Mercurial => Ok(Box::new(HgScm::init(path)?)),
187    }
188}
189
190/// Detect which backend a repository uses.
191///
192/// Returns `None` if the path is not a repository.
193pub fn detect_backend(path: &Path) -> Option<Backend> {
194    if path.join(".git").exists() {
195        Some(Backend::Git)
196    } else if path.join(".hg").exists() {
197        Some(Backend::Mercurial)
198    } else {
199        None
200    }
201}
