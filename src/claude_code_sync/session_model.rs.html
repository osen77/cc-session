11/// Identifies the system that produced a session.
12#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, serde::Serialize, serde::Deserialize)]
13#[serde(rename_all = "lowercase")]
14pub enum SessionSource {
15    /// Claude Code session.
16    Claude,
17    /// Codex session.
18    Codex,
19    /// Oh My Pi session.
20    Omp,
21    /// Google Antigravity CLI (`agy`) session.
22    Agy,
23}

25/// Operations supported by a session source.
26#[derive(Debug, Clone, Copy, PartialEq, Eq)]
27pub struct SourceCapabilities {
28    /// Whether the session can be opened in its source application.
29    pub can_open: bool,
30    /// Whether the session can be renamed.
31    pub can_rename: bool,
32    /// Whether the session can be deleted.
33    pub can_delete: bool,
34    /// Whether the session participates in synchronization.
35    pub participates_in_sync: bool,
36}

38impl SessionSource {
39    /// Returns the stable lowercase source identifier.
40    pub fn as_str(self) -> &'static str {
41        match self {
42            Self::Claude => "claude",
43            Self::Codex => "codex",
44            Self::Omp => "omp",
45            Self::Agy => "agy",
46        }
47    }

49    /// Returns the short display label used by session listings.
50    pub fn label(self) -> &'static str {
51        match self {
52            Self::Claude => "CC",
53            Self::Codex => "CX",
54            Self::Omp => "OM",
55            Self::Agy => "AG",
56        }
57    }

59    /// Returns the operations supported by this source.
60    pub fn capabilities(self) -> SourceCapabilities {
61        match self {
62            Self::Claude => SourceCapabilities {
63                can_open: true,
64                can_rename: true,
65                can_delete: true,
66                participates_in_sync: true,
67            },
68            Self::Codex => SourceCapabilities {
69                can_open: false,
70                can_rename: false,
71                can_delete: false,
72                participates_in_sync: false,
73            },
74            Self::Omp => SourceCapabilities {
75                can_open: true,
76                can_rename: false,
77                can_delete: false,
78                participates_in_sync: false,
79            },
80            Self::Agy => SourceCapabilities {
81                can_open: true,
82                can_rename: false,
83                can_delete: false,
84                participates_in_sync: false,
85            },
86        }

113/// Filter selecting one or all session sources.
114#[derive(Debug, Clone, Copy, PartialEq, Eq)]
115pub enum SessionSourceFilter {
116    /// Include sessions from every supported source.
117    All,
118    /// Include only Claude Code sessions.
119    Claude,
120    /// Include only Codex sessions.
121    Codex,
122    /// Include only Oh My Pi sessions.
123    Omp,
124    /// Include only Google Antigravity CLI (`agy`) sessions.
125    Agy,
126}
127
128impl SessionSourceFilter {
129    pub(crate) fn includes(self, source: SessionSource) -> bool {
130        matches!(
131            (self, source),
132            (Self::All, _)
133                | (Self::Claude, SessionSource::Claude)
134                | (Self::Codex, SessionSource::Codex)
135                | (Self::Omp, SessionSource::Omp)
136                | (Self::Agy, SessionSource::Agy)
137        )
138    }
139
140    pub(crate) fn includes_claude(self) -> bool {
141        matches!(self, Self::All | Self::Claude)
142    }
143
144    pub(crate) fn includes_codex(self) -> bool {
145        matches!(self, Self::All | Self::Codex)
146    }
147
148    pub(crate) fn includes_omp(self) -> bool {
149        matches!(self, Self::All | Self::Omp)
150    }
151
152    pub(crate) fn includes_agy(self) -> bool {
153        matches!(self, Self::All | Self::Agy)
154    }
155
156    /// Whether this filter selects any source backed by `session_index.json`.
157    pub(crate) fn uses_index_cache(self) -> bool {
158        self.includes_claude() || self.includes_codex() || self.includes_omp()
159    }
160}

175/// Session summary for listing and operations.

177pub struct SessionSummary {
178    /// Stable source identifier retained for cache and JSON compatibility.
179    pub source: String,
180    /// Source-local session identifier.
181    pub session_id: String,
182    /// User-visible session title.
183    pub title: String,
184    /// Name of the project associated with the session.
185    pub project_name: String,
186    /// Filesystem directory associated with the project.
187    ///
188    /// For Claude this is the encoded `~/.claude/projects/<encoded>` storage
189    /// directory, not the directory the session ran in. Use [`SessionSummary::cwd`]
190    /// when the real working directory matters.
191    pub project_dir: PathBuf,
192    /// Real working directory the session ran in, when the source records one.
193    pub cwd: Option<String>,
194    /// Filesystem path of the source session file.
195    pub file_path: PathBuf,
196    /// Total number of user and assistant turns.
197    pub message_count: usize,
198    /// Number of user turns in the session.
199    pub user_message_count: usize,
200    /// Number of assistant turns in the session.
201    pub assistant_message_count: usize,
202    /// Timestamp of the first recorded message, if available.
203    pub first_timestamp: Option<String>,
204    /// Timestamp of the most recent activity, if available.
205    pub last_activity: Option<String>,
206    /// Size of the source session file in bytes.
207    pub file_size: u64,
208    /// Whether a user-created custom title protects this session from title replacement.
209    pub has_custom_title: bool,
210}

212impl SessionSummary {
213    /// Create a SessionSummary from a ConversationSession.
214    /// Message counts use "turn" granularity: consecutive assistant entries
215    /// between two user messages count as one assistant turn.
216    pub fn from_session(
217        session: &ConversationSession,
218        project_name: &str,
219        project_dir: &Path,
220    ) -> Self {
221        let file_size = fs::metadata(&session.file_path)
222            .map(|m| m.len())
223            .unwrap_or(0);
224
225        let mut user_count = 0;
226        let mut assistant_count = 0;
227        let mut in_assistant_turn = false;
228
229        for entry in &session.entries {
230            match entry.entry_type.as_str() {
231                "user" => {
232                    if ConversationSession::is_tool_result_entry(entry) {
233                        continue;
234                    }
235                    user_count += 1;
236                    in_assistant_turn = false;
237                }
238                "assistant" => {
239                    if !in_assistant_turn {
240                        assistant_count += 1;
241                        in_assistant_turn = true;
242                    }
243                }
244                _ => {}
245            }
246        }
247
248        SessionSummary {
249            source: SessionSource::Claude.as_str().to_string(),
250            session_id: session.session_id.clone(),
251            title: session.title().unwrap_or_else(|| "(No title)".to_string()),
252            project_name: project_name.to_string(),
253            project_dir: project_dir.to_path_buf(),
254            cwd: session.cwd().map(str::to_string),
255            file_path: PathBuf::from(&session.file_path),
256            message_count: user_count + assistant_count,
257            user_message_count: user_count,
258            assistant_message_count: assistant_count,
259            first_timestamp: session.first_timestamp(),
260            last_activity: session.latest_timestamp(),
261            file_size,
262            has_custom_title: session.has_custom_title(),
263        }
264    }

267    ///
268    /// This semantic check is shared by active filesystem scans and recycled-file
269    /// queries so parseable metadata-only or title-less sessions are never exposed.
270    pub(crate) fn is_valid(&self) -> bool {
271        self.message_count > 0 && !self.title.trim().is_empty() && self.title != "(No title)"
272    }

274    /// Get a truncated title for display (Unicode-safe).
275    pub fn display_title(&self, max_chars: usize) -> String {
276        let title = self.title.replace('\n', " ");
277        let chars: Vec<char> = title.chars().collect();
278
279        if chars.len() > max_chars {
280            let truncated: String = chars[..max_chars - 3].iter().collect();
281            format!("{}...", truncated)
282        } else {
283            title
284        }
285    }

382    /// Create a SessionSummary from an `agy` transcript, optionally enriched with SQLite metadata.
383    pub fn from_agy_session(
384        session: &AgySession,
385        show_path: PathBuf,
386        db: Option<&AgyDbSummary>,
387    ) -> Self {
388        let file_size = fs::metadata(&show_path).map(|m| m.len()).unwrap_or(0);
389        let messages = session.display_messages();
390        let user_count = messages.iter().filter(|m| m.role == "user").count();
391        let assistant_count = messages.iter().filter(|m| m.role == "assistant").count();
392        let title = session.title_from_messages(
393            &messages,
394            db.map(|d| d.title.as_str()),
395            db.map(|d| d.preview.as_str()),
396        );
397        let cwd = db
398            .and_then(|d| d.cwd.clone())
399            .or_else(|| session.cwd.clone());
400        let project_name = db
401            .map(|d| d.project_name.clone())
402            .or_else(|| session.project_name().map(str::to_string))
403            .unwrap_or_else(|| "agy".to_string());
404        let project_dir = cwd.as_deref().map(PathBuf::from).unwrap_or_default();
405        let last_activity = match (
406            db.and_then(|d| d.last_modified_time.clone()),
407            session.latest_timestamp(),
408        ) {
409            (Some(a), Some(b)) => Some(a.max(b)),
410            (Some(a), None) => Some(a),
411            (None, b) => b,
412        };
413        let first_timestamp = session
414            .first_timestamp()
415            .or_else(|| db.and_then(|d| d.last_user_input_time.clone()))
416            .or_else(|| last_activity.clone());
417
418        SessionSummary {
419            source: SessionSource::Agy.as_str().to_string(),
420            session_id: session.session_id.clone(),
421            title,
422            project_name,
423            project_dir,
424            cwd,
425            file_path: show_path,
426            message_count: user_count + assistant_count,
427            user_message_count: user_count,
428            assistant_message_count: assistant_count,
429            first_timestamp,
430            last_activity,
431            file_size,
432            has_custom_title: false,
433        }
434    }

436    /// Create a SessionSummary from an `agy` SQLite metadata row when no transcript file exists.
437    pub fn from_agy_db_only(db: &AgyDbSummary, db_path: &Path) -> Self {
438        let title = if !db.title.is_empty() {
439            db.title.clone()
440        } else if !db.preview.is_empty() {
441            db.preview.clone()
442        } else {
443            "(No title)".to_string()
444        };
445        let has_content = !db.preview.is_empty() || !db.title.is_empty();
446        let (user_count, message_count) = if has_content && db.parent_conversation_id.is_empty() {
447            (1, db.step_count.max(1))
448        } else {
449            (0, 0)
450        };
451        let assistant_count = message_count.saturating_sub(user_count);
452        let project_dir = db.cwd.as_deref().map(PathBuf::from).unwrap_or_default();
453        let first_timestamp = db
454            .last_user_input_time
455            .clone()
456            .or_else(|| db.last_modified_time.clone());
457
458        SessionSummary {
459            source: SessionSource::Agy.as_str().to_string(),
460            session_id: db.conversation_id.clone(),
461            title,
462            project_name: db.project_name.clone(),
463            project_dir,
464            cwd: db.cwd.clone(),
465            file_path: db_path.to_path_buf(),
466            message_count,
467            user_message_count: user_count,
468            assistant_message_count: assistant_count,
469            first_timestamp,
470            last_activity: db.last_modified_time.clone(),
471            file_size: 0,
472            has_custom_title: false,
473        }
474    }

477/// Format a timestamp as relative time (e.g., "Today", "Yesterday", "3 days ago").
478pub(crate) fn format_relative_time(timestamp: &str) -> String {
479    use chrono::{DateTime, Utc};
480
481    if let Ok(dt) = DateTime::parse_from_rfc3339(timestamp) {
482        let now = Utc::now();
483        let dt_utc = dt.with_timezone(&Utc);
484        let duration = now.signed_duration_since(dt_utc);
485
486        let days = duration.num_days();
487        let hours = duration.num_hours();
488        let minutes = duration.num_minutes();
489
490        if days == 0 {
491            if hours == 0 {
492                if minutes <= 1 {
493                    "Just now".to_string()
494                } else {
495                    format!("{} min ago", minutes)
496                }
497            } else if hours == 1 {
498                "1 hour ago".to_string()
499            } else {
500                format!("{} hours ago", hours)
501            }
502        } else if days == 1 {
503            "Yesterday".to_string()
504        } else if days < 7 {
505            format!("{} days ago", days)
506        } else if days < 30 {
507            let weeks = days / 7;
508            if weeks == 1 {
509                "1 week ago".to_string()
510            } else {
511                format!("{} weeks ago", weeks)
512            }
513        } else {
514            let months = days / 30;
515            if months == 1 {
516                "1 month ago".to_string()
517            } else {
518                format!("{} months ago", months)
519            }
520        }
521    } else {
522        "Unknown".to_string()
523    }
524}
