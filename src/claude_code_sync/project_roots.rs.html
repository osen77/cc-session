25/// Local-only authorization for the entire Claude projects root.
26///
27/// The logical `~/.claude/projects` directory remains the stable identity while
28/// all physical I/O is redirected to `target` after the volume and link are
29/// revalidated. This is intentionally separate from the legacy per-project
30/// `project_roots` array.

32#[serde(deny_unknown_fields)]
33pub struct ExternalProjectsRoot {
34    pub target: PathBuf,
35    pub trusted_root: PathBuf,
36    pub volume_uuid: String,
37}

122/// One operation's validated configuration and project inventory. Resolutions
123/// still probe the selected mapping on every call; this is not a volume cache.
124pub(crate) struct ProjectRootIndex<'a> {
125    projects_root: &'a Path,
126    mappings: std::collections::HashMap<&'a str, &'a ProjectRootMapping>,
127    roots: Vec<ResolvedProjectRoot>,
128}
129
130impl<'a> ProjectRootIndex<'a> {
131    pub(crate) fn new(projects_root: &'a Path, mappings: &'a [ProjectRootMapping]) -> Result<Self> {
132        Self::with_probe(projects_root, mappings, &volume_probe)
133    }
134
135    fn with_probe(projects_root: &'a Path, mappings: &'a [ProjectRootMapping], probe: &dyn VolumeProbe) -> Result<Self> {
136        let roots = enumerate_with_probe(projects_root, mappings, probe)?;
137        Ok(Self { projects_root, mappings: mappings.iter().map(|m| (m.project_dir.as_str(), m)).collect(), roots })
138    }
139
140    pub(crate) fn roots(&self) -> &[ResolvedProjectRoot] { &self.roots }
141
142    pub(crate) fn resolve(&self, logical_project: &Path) -> Result<ResolvedProjectRoot> {
143        self.resolve_with_probe(logical_project, &volume_probe)
144    }
145
146    fn resolve_with_probe(&self, logical_project: &Path, probe: &dyn VolumeProbe) -> Result<ResolvedProjectRoot> {
147        let name = logical_project.strip_prefix(self.projects_root)?.to_str().context("project is not UTF-8")?;
148        validate_project_component(name)?;
149        let position = self.roots.binary_search_by(|root| root.project_dir.as_str().cmp(name))
150            .map_err(|_| anyhow!("project was not present at operation preflight"))?;
151        let expected = &self.roots[position];
152        if let Some(mapping) = self.mappings.get(name) {
153            validate_directory_root(self.projects_root)?;
154            strict_absolute_directory(self.projects_root)?;
155            let observed = validate_mapping(self.projects_root, mapping, probe)?;
156            anyhow::ensure!(observed == *expected, "project mapping changed during operation");
157            Ok(observed)
158        } else if let Some(mapping) = self.mappings.get(EXTERNAL_ROOT_PROJECT_DIR) {
159            let observed = validate_external_project(self.projects_root, mapping, name, probe)?;
160            anyhow::ensure!(observed == *expected, "external project root changed during operation");
161            Ok(observed)
162        } else {
163            validate_directory_root(self.projects_root)?;
164            validate_directory_candidate(self.projects_root, logical_project)?;
165            Ok(expected.clone())
166        }
167    }
168
169    pub(crate) fn file_boundary(&self, relative: &Path) -> Result<(PathBuf, PathBuf)> {
170        let mut components = relative.components();
171        let Some(Component::Normal(project)) = components.next() else { bail!("invalid logical project path"); };
172        let root = self.resolve(&self.projects_root.join(project))?;
173        let tail = components.as_path().to_path_buf();
174        crate::path_security::safe_join_within_root(&root.physical_root, &tail)?;
175        Ok((root.physical_root, tail))
176    }
177
178    pub(crate) fn session_project_root(&self, session: &Path) -> Result<ResolvedProjectRoot> {
179        let relative = session.strip_prefix(self.projects_root)?;
180        let mut components = relative.components();
181        let Some(Component::Normal(project)) = components.next() else { bail!("invalid session project component"); };
182        let root = self.resolve(&self.projects_root.join(project))?;
183        let physical = crate::path_security::safe_join_within_root(&root.physical_root, components.as_path())?;
184        crate::path_security::validate_regular_candidate(&root.physical_root, &physical)?;
185        Ok(root)
186    }
187}
188/// Resolve a logical file destination without creating anything. New ordinary
189/// projects keep the original root boundary; registered missing projects fail.
190pub(crate) fn file_boundary(
191    projects_root: &Path,
192    mappings: &[ProjectRootMapping],
193    relative: &Path,
194) -> Result<(PathBuf, PathBuf)> {
195    let roots = enumerate(projects_root, mappings)?;
196    let mut components = relative.components();
197    let Some(Component::Normal(project)) = components.next() else {
198        bail!("invalid logical project path");
199    };
200    let tail = components.as_path();
201    let root = roots
202        .iter()
203        .find(|root| std::ffi::OsStr::new(&root.project_dir) == project);
204    let (boundary, relative) = match root {
205        Some(root) => (root.physical_root.clone(), tail.to_path_buf()),
206        None => {
207            if let Some(external) = mappings
208                .iter()
209                .find(|mapping| mapping.project_dir == EXTERNAL_ROOT_PROJECT_DIR)
210            {
211                (external.target.clone(), relative.to_path_buf())
212            } else {
213                (projects_root.to_path_buf(), relative.to_path_buf())
214            }
215        }
216    };
217    crate::path_security::safe_join_within_root(&boundary, &relative)?;
218    Ok((boundary, relative))
219}

222pub fn resolve_with_probe(
223    projects_root: &Path,
224    mappings: &[ProjectRootMapping],
225    logical_project: &Path,
226    probe: &dyn VolumeProbe,
227) -> Result<ResolvedProjectRoot> {
228    let relative = logical_project
229        .strip_prefix(projects_root)
230        .context("logical project must be below the Claude projects root")?;
231    let name = relative
232        .to_str()
233        .ok_or_else(|| anyhow!("project name is not UTF-8"))?;
234    validate_project_component(name)?;
235    // Enumeration also rejects unknown links before any consumer can mutate state.
236    enumerate_with_probe(projects_root, mappings, probe)?
237        .into_iter()
238        .find(|root| root.project_dir == name)
239        .ok_or_else(|| {
240            anyhow!(
241                "Claude project does not exist: {}",
242                logical_project.display()
243            )
244        })
245}

247pub fn enumerate_with_probe(
248    projects_root: &Path,
249    mappings: &[ProjectRootMapping],
250    probe: &dyn VolumeProbe,
251) -> Result<Vec<ResolvedProjectRoot>> {
252    crate::sync::push_diagnostics::enumerate();
253    let external = mappings
254        .iter()
255        .find(|mapping| mapping.project_dir == EXTERNAL_ROOT_PROJECT_DIR);
256    if let Some(external) = external {
257        if mappings.len() != 1 {
258            bail!("external projects root cannot be mixed with project mappings");
259        }
260        return enumerate_external_root(projects_root, external, probe);
261    }
262
263    validate_directory_root(projects_root)?;
264    if !mappings.is_empty() {
265        strict_absolute_directory(projects_root)?;
266    }
267    let mut names = HashSet::with_capacity(mappings.len());
268    let mut targets: Vec<PathBuf> = Vec::with_capacity(mappings.len());
269    let mut roots = Vec::new();
270    // Validate every mapping, including entries absent from read_dir, first.
271    for mapping in mappings {
272        validate_project_component(&mapping.project_dir)?;
273        if !names.insert(mapping.project_dir.as_str()) {
274            bail!("duplicate project root mapping: {}", mapping.project_dir);
275        }
276        let root = validate_mapping(projects_root, mapping, probe)?;
277        let canonical = fs::canonicalize(&root.physical_root)?;
278        if targets.iter().any(|target| canonical.starts_with(target) || target.starts_with(&canonical)) {
279            bail!("overlapping canonical project root mappings: {}", mapping.project_dir);
280        }
281        targets.push(canonical);
282        roots.push(root);
283    }
284    for entry in fs::read_dir(projects_root).context("failed to enumerate Claude projects")? {
285        let entry = entry?;
286        let file_type = entry.file_type()?;
287        let name = entry.file_name();
288        if names.contains(name.to_str().unwrap_or_default()) {
289            continue;
290        }
291        if file_type.is_symlink() {
292            bail!(
293                "unregistered Claude project symlink: {}",
294                entry.path().display()
295            );
296        }
297        if !file_type.is_dir() {
298            continue;
299        }
300        let project_dir = name
301            .into_string()
302            .map_err(|_| anyhow!("project name is not UTF-8"))?;
303        validate_project_component(&project_dir)?;
304        validate_directory_candidate(projects_root, &entry.path())?;
305        roots.push(ResolvedProjectRoot {
306            project_dir,
307            logical_root: entry.path(),
308            physical_root: entry.path(),
309            is_mapped: false,
310        });
311    }
312    roots.sort_by(|a, b| a.project_dir.cmp(&b.project_dir));
313    Ok(roots)
314}

316fn strict_absolute_directory(path: &Path) -> Result<()> {
317    if !path.is_absolute()
318        || path
319            .components()
320            .any(|c| !matches!(c, Component::RootDir | Component::Normal(_)))
321    {
322        bail!(
323            "mapped directory must be an absolute, normalized path: {}",
324            path.display()
325        );
326    }
327    // Check from the filesystem root: no ancestor may redirect an authorization.
328    let mut current = PathBuf::new();
329    for component in path.components() {
330        current.push(component.as_os_str());
331        validate_directory_root(&current)?;
332    }
333    Ok(())
334}

336fn enumerate_external_root(
337    projects_root: &Path,
338    mapping: &ProjectRootMapping,
339    probe: &dyn VolumeProbe,
340) -> Result<Vec<ResolvedProjectRoot>> {
341    validate_external_root_link(projects_root, mapping, probe)?;
342    let mut roots = Vec::new();
343    for entry in fs::read_dir(&mapping.target).context("failed to enumerate external Claude projects")? {
344        let entry = entry?;
345        let file_type = entry.file_type()?;
346        if file_type.is_symlink() {
347            bail!("external Claude project symlink is not allowed: {}", entry.path().display());
348        }
349        if !file_type.is_dir() {
350            continue;
351        }
352        let project_dir = entry
353            .file_name()
354            .into_string()
355            .map_err(|_| anyhow!("external project name is not UTF-8"))?;
356        validate_project_component(&project_dir)?;
357        validate_directory_candidate(&mapping.target, &entry.path())?;
358        roots.push(ResolvedProjectRoot {
359            project_dir: project_dir.clone(),
360            logical_root: projects_root.join(&project_dir),
361            physical_root: entry.path(),
362            is_mapped: true,
363        });
364    }
365    roots.sort_by(|a, b| a.project_dir.cmp(&b.project_dir));
366    Ok(roots)
367}

369fn validate_external_project(
370    projects_root: &Path,
371    mapping: &ProjectRootMapping,
372    project_dir: &str,
373    probe: &dyn VolumeProbe,
374) -> Result<ResolvedProjectRoot> {
375    validate_external_root_link(projects_root, mapping, probe)?;
376    let logical_root = projects_root.join(project_dir);
377    let physical_root = mapping.target.join(project_dir);
378    validate_directory_candidate(&mapping.target, &physical_root)?;
379    Ok(ResolvedProjectRoot {
380        project_dir: project_dir.to_owned(),
381        logical_root,
382        physical_root,
383        is_mapped: true,
384    })
385}

387fn validate_external_root_link(
388    projects_root: &Path,
389    mapping: &ProjectRootMapping,
390    probe: &dyn VolumeProbe,
391) -> Result<()> {
392    let metadata = fs::symlink_metadata(projects_root)
393        .with_context(|| format!("failed to inspect Claude projects root: {}", projects_root.display()))?;
394    if !metadata.file_type().is_symlink() {
395        bail!("external projects root must be a single-hop symlink: {}", projects_root.display());
396    }
397    strict_absolute_directory(&mapping.trusted_root)?;
398    strict_absolute_directory(&mapping.target)?;
399    if mapping.target == mapping.trusted_root || !mapping.target.starts_with(&mapping.trusted_root) {
400        bail!("external projects target must remain below trusted root");
401    }
402    let link_target = fs::read_link(projects_root)?;
403    if !link_target.is_absolute() || link_target != mapping.target {
404        bail!("external projects root symlink target changed or is not absolute");
405    }
406    validate_directory_candidate(&mapping.trusted_root, &mapping.target)?;
407    let expected_uuid = uuid::Uuid::parse_str(&mapping.volume_uuid).context("invalid external volume UUID")?;
408    if expected_uuid.is_nil() {
409        bail!("nil external volume UUID is not allowed");
410    }
411    crate::sync::push_diagnostics::probe();
412    let volume = probe.probe(&mapping.trusted_root)?;
413    strict_absolute_directory(&volume.mount_point)?;
414    if volume.mount_point.parent().is_none()
415        || !mapping.trusted_root.starts_with(&volume.mount_point)
416        || uuid::Uuid::parse_str(&volume.volume_uuid).context("invalid probed volume UUID")? != expected_uuid
417    {
418        bail!("external volume mount point or UUID does not match");
419    }
420    let relative = mapping.target.strip_prefix(&volume.mount_point)?;
421    let mut current = volume.mount_point.clone();
422    ensure_device(&current, volume.device_id)?;
423    for component in relative.components() {
424        current.push(component.as_os_str());
425        ensure_device(&current, volume.device_id)?;
426    }
427    if fs::read_link(projects_root)? != mapping.target {
428        bail!("external projects root symlink changed during volume validation");
429    }
430    Ok(())
431}

433fn validate_mapping(
434    projects_root: &Path,
435    mapping: &ProjectRootMapping,
436    probe: &dyn VolumeProbe,
437) -> Result<ResolvedProjectRoot> {
438    let expected_uuid =
439        uuid::Uuid::parse_str(&mapping.volume_uuid).context("invalid mapped volume UUID")?;
440    if expected_uuid.is_nil() {
441        bail!("nil mapped volume UUID is not allowed");
442    }
443    strict_absolute_directory(&mapping.trusted_root)?;
444    strict_absolute_directory(&mapping.target)?;
445    if mapping.target == mapping.trusted_root
446        || !mapping.target.starts_with(&mapping.trusted_root)
447        || mapping.target.file_name() != Some(std::ffi::OsStr::new(&mapping.project_dir))
448    {
449        bail!("mapped target must retain its project name below the trusted root");
450    }
451    validate_directory_candidate(&mapping.trusted_root, &mapping.target)?;
452    let logical_root = projects_root.join(&mapping.project_dir);
453    if !fs::symlink_metadata(&logical_root)?
454        .file_type()
455        .is_symlink()
456    {
457        bail!(
458            "registered project must be a single-hop symlink: {}",
459            logical_root.display()
460        );
461    }
462    if fs::read_link(&logical_root)? != mapping.target {
463        bail!(
464            "registered project symlink target changed: {}",
465            logical_root.display()
466        );
467    }
468    crate::sync::push_diagnostics::probe();
469    let volume = probe.probe(&mapping.trusted_root)?;
470    strict_absolute_directory(&volume.mount_point)?;
471    if volume.mount_point.parent().is_none()
472        || !mapping.trusted_root.starts_with(&volume.mount_point)
473        || uuid::Uuid::parse_str(&volume.volume_uuid).context("invalid probed volume UUID")?
474            != expected_uuid
475    {
476        bail!("mapped volume mount point or UUID does not match");
477    }
478    let relative = mapping.target.strip_prefix(&volume.mount_point)?;
479    let mut current = volume.mount_point.clone();
480    ensure_device(&current, volume.device_id)?;
481    for component in relative.components() {
482        current.push(component.as_os_str());
483        ensure_device(&current, volume.device_id)?;
484    }
485    // Recheck the link after probing (which can involve a bounded subprocess).
486    if fs::read_link(&logical_root)? != mapping.target {
487        bail!("registered project symlink changed during volume validation");
488    }
489    Ok(ResolvedProjectRoot {
490        project_dir: mapping.project_dir.clone(),
491        logical_root,
492        physical_root: mapping.target.clone(),
493        is_mapped: true,
494    })
495}

497fn ensure_device(path: &Path, expected: u64) -> Result<()> {
498    let metadata = fs::symlink_metadata(path)?;
499    if !metadata.is_dir() || metadata.file_type().is_symlink() || device_id(&metadata)? != expected
500    {
501        bail!(
502            "mapped path is not a real directory on the authorized device: {}",
503            path.display()
504        );
505    }
506    Ok(())
507}

525#[cfg(target_os = "macos")]
526fn system_volume_probe(path: &Path) -> Result<VolumeIdentity> {
527    use std::os::unix::fs::MetadataExt;
528    use std::process::Command;
529    strict_absolute_directory(path)?;
530    let expected_device = fs::metadata(path)?.dev();
531    let mut mount_point = path.to_path_buf();
532    loop {
533        let parent = mount_point.parent().ok_or_else(|| anyhow!("no non-root mount boundary for mapped directory"))?;
534        if fs::metadata(parent)?.dev() != expected_device { break; }
535        mount_point = parent.to_path_buf();
536    }
537    let mut diskutil = Command::new("/usr/sbin/diskutil");
538    diskutil.args(["info", "-plist"]).arg(&mount_point);
539    crate::sync::push_diagnostics::diskutil();
540    let plist = bounded_output(diskutil, None)?;
541    // Apple's parser handles XML escaping and plist types; no ad-hoc XML parsing.
542    let mut plutil = Command::new("/usr/bin/plutil");
543    plutil.args(["-convert", "json", "-o", "-", "--", "-"]);
544    let json = bounded_output(plutil, Some(plist))?;
545    let info: serde_json::Value =
546        serde_json::from_slice(&json).context("invalid diskutil plist")?;
547    let text = |key: &str| -> Result<&str> {
548        info.get(key)
549            .and_then(|v| v.as_str())
550            .filter(|v| !v.is_empty())
551            .ok_or_else(|| anyhow!("diskutil omitted {key}"))
552    };
553    validate_diskutil_mount(&info, &mount_point)?;
554    strict_absolute_directory(&mount_point)?;
555    if !path.starts_with(&mount_point) || mount_point.parent().is_none() {
556        bail!("diskutil did not identify an external mounted filesystem");
557    }
558    let device_node = PathBuf::from(text("DeviceNode")?);
559    if device_node.parent() != Some(Path::new("/dev")) {
560        bail!("diskutil returned an invalid device node");
561    }
562    let device = fs::symlink_metadata(&device_node)?;
563    use std::os::unix::fs::FileTypeExt;
564    if !device.file_type().is_block_device() {
565        bail!("volume device is not a block device");
566    }
567    let mounted = fs::metadata(&mount_point)?;
568    if mounted.dev() != expected_device { bail!("mapped device changed during volume verification"); }
569    let mut component_path = mount_point.clone();
570    for component in path.strip_prefix(&mount_point)?.components() {
571        component_path.push(component.as_os_str());
572        ensure_device(&component_path, expected_device)?;
573    }
574    if device.rdev() != mounted.dev() {
575        bail!("diskutil device and mounted filesystem differ");
576    }
577    let parent = fs::metadata(mount_point.parent().unwrap())?;
578    if parent.dev() == mounted.dev() {
579        bail!("reported mount point is not a filesystem boundary");
580    }
581    Ok(VolumeIdentity {
582        mount_point,
583        volume_uuid: text("VolumeUUID")?.to_owned(),
584        device_id: mounted.dev(),
585    })
586}

588#[cfg(target_os = "macos")]
589fn validate_diskutil_mount(info: &serde_json::Value, mount: &Path) -> Result<()> {
590    if info.get("Mounted").is_some_and(|value| value.as_bool() != Some(true)) {
591        bail!("diskutil reports volume is not mounted");
592    }
593    if info.get("MountPoint").and_then(|value| value.as_str()).map(Path::new) != Some(mount) {
594        bail!("diskutil mount point differs from observed device boundary");
595    }
596    if info.get("FilesystemType").and_then(|value| value.as_str()).is_none_or(|kind| !kind.eq_ignore_ascii_case("apfs")) {
597        bail!("mapped volume must be APFS");
598    }
599    for key in ["Writable", "WritableVolume", "WritableMedia"] {
600        if info.get(key).is_some_and(|value| value.as_bool() != Some(true)) {
601            bail!("mapped volume is not writable");
602        }
603    }
604    let uuid = info.get("VolumeUUID").and_then(|value| value.as_str()).ok_or_else(|| anyhow!("diskutil omitted VolumeUUID"))?;
605    if uuid::Uuid::parse_str(uuid)?.is_nil() { bail!("nil volume UUID"); }
606    Ok(())
607}

609/// Both process duration and output are bounded; commands are fixed system tools.
610#[cfg(target_os = "macos")]
611fn bounded_output(mut command: std::process::Command, input: Option<Vec<u8>>) -> Result<Vec<u8>> {
612    use std::io::{Read, Write};
613    use std::process::Stdio;
614    use std::sync::mpsc;
615    use std::time::{Duration, Instant};
616    const LIMIT: usize = 128 * 1024;
617    command
618        .stdin(if input.is_some() {
619            Stdio::piped()
620        } else {
621            Stdio::null()
622        })
623        .stdout(Stdio::piped())
624        .stderr(Stdio::null());
625    let mut child = command
626        .spawn()
627        .context("failed to start volume verification tool")?;
628    if let Some(input) = input {
629        let mut stdin = child.stdin.take().unwrap();
630        std::thread::spawn(move || {
631            let _ = stdin.write_all(&input);
632        });
633    }
634    let stdout = child.stdout.take().unwrap();
635    let (sender, receiver) = mpsc::channel();
636    std::thread::spawn(move || {
637        let mut bytes = Vec::new();
638        let result = stdout
639            .take((LIMIT + 1) as u64)
640            .read_to_end(&mut bytes)
641            .map(|_| bytes);
642        let _ = sender.send(result);
643    });
644    let deadline = Instant::now() + Duration::from_secs(5);
645    let result = (|| -> Result<Vec<u8>> {
646        let bytes = receiver
647            .recv_timeout(deadline.saturating_duration_since(Instant::now()))
648            .context("volume verification timed out")??;
649        if bytes.len() > LIMIT {
650            bail!("volume verification output exceeds limit");
651        }
652        loop {
653            if let Some(status) = child.try_wait()? {
654                if !status.success() {
655                    bail!("volume verification tool failed: {status}");
656                }
657                return Ok(bytes);
658            }
659            if Instant::now() >= deadline {
660                bail!("volume verification timed out");
661            }
662            std::thread::sleep(Duration::from_millis(10));
663        }
664    })();
665    if result.is_err() {
666        let _ = child.kill();
667        let _ = child.wait();
668    }
669    result
670}
